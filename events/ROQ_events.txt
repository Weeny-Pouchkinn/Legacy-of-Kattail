#If controller of Solkatzia is at risk of surrender, they ask Roqualia if they can join!
country_event = {
	id = ROQ_event.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event ROQ_event.1"}
	title = ROQ_event.1.t
	desc = ROQ_event.1.d
	picture = GFX_report_event_dofus

	trigger = {
		tag = ROQ
		is_puppet = no
		514 = {
			var:controller = {
				is_herzlands_warlord = yes
				has_democratic_gov_group = yes
				surrender_progress > 0.1
				OR = { 
					has_war_with = AUR
					has_war_with = SIL
					has_war_with = PAW
					has_war_with = CLE
				}
				NOT = { has_war_with = ROQ }
			}
		}
	}

	fire_only_once = yes

	option = {
		name = ROQ_event.1.a
		custom_effect_tooltip = AUR_event.1.a.tt
		effect_tooltip = {
			514 = { var:controller = { add_ideas = ROQ_south_looks_to_roqualia } }
		}
		add_political_power = -100

		514 = {
			add_claim_by = ROQ
		}
		512 = {
			add_claim_by = ROQ
		}
		191 = {
			add_claim_by = ROQ
		}

		every_country = {
			limit = { has_war_with = 514.owner }
			country_event = { id = ROQ_event.2 days = 1 }
		}

		514 = {
			var:owner = {
				country_event = { id = ROQ_event.3 days = 1 }
			}
		}
	}

	option = {
		name = ROQ_event.1.b
		add_political_power = 50
		ai_chance = { base = 0 }
	}
}

country_event = {
	id = ROQ_event.5
	immediate = { log = "[GetDateText]: [Root.GetName]: event ROQ_event.5"}
	title = ROQ_event.5.t
	desc = ROQ_event.5.d
	picture = GFX_report_event_dofus

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			random_list = {
				1 = {}
				1 = {
					GLI = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = GLI }
					modifier = { factor = 0 has_global_flag = roq_splinter_GLI_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = GLI } } } }
				}
				1 = {
					UCE = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = UCE }
					modifier = { factor = 0 has_global_flag = roq_splinter_UCE_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = UCE } } } }
				}
				1 = {
					VEL = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = VEL }
					modifier = { factor = 0 has_global_flag = roq_splinter_VEL_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = VEL } } } }
				}
				1 = {
					REC = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = REC }
					modifier = { factor = 0 has_global_flag = roq_splinter_REC_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = REC } } } }
				}
				1 = {
					POI = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = POI }
					modifier = { factor = 0 has_global_flag = roq_splinter_POI_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = POI } } } }
				}
				1 = {
					OBS = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = OBS }
					modifier = { factor = 0 has_global_flag = roq_splinter_OBS_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = OBS } } } }
				}
				1 = {
					DEM = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = DEM }
					modifier = { factor = 0 has_global_flag = roq_splinter_DEM_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = DEM } } } }
				}
				1 = {
					BRU = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = BRU }
					modifier = { factor = 0 has_global_flag = roq_splinter_BRU_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = BRU } } } }
				}
				1 = {
					BOC = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = BOC }
					modifier = { factor = 0 has_global_flag = roq_splinter_BOC_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = BOC } } } }
				}
				1 = {
					PER = { save_global_event_target_as = roq_revolt }
					modifier = { factor = 0 country_exists = PER }
					modifier = { factor = 0 has_global_flag = roq_splinter_PER_revolted }
					modifier = { factor = 0 ROQ = { NOT = { any_controlled_state = { is_core_of = PER } } } }
				}
			}
			# Mark this splinter as having revolted (can only revolt once)
			if = {
				limit = { event_target:roq_revolt = { tag = GLI } }
				set_global_flag = roq_splinter_GLI_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = UCE } }
				set_global_flag = roq_splinter_UCE_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = VEL } }
				set_global_flag = roq_splinter_VEL_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = REC } }
				set_global_flag = roq_splinter_REC_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = POI } }
				set_global_flag = roq_splinter_POI_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = OBS } }
				set_global_flag = roq_splinter_OBS_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = DEM } }
				set_global_flag = roq_splinter_DEM_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = BRU } }
				set_global_flag = roq_splinter_BRU_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = BOC } }
				set_global_flag = roq_splinter_BOC_revolted
			}
			else_if = {
				limit = { event_target:roq_revolt = { tag = PER } }
				set_global_flag = roq_splinter_PER_revolted
			}
			ROQ = {
				random_owned_controlled_state = {
					limit = { is_core_of = event_target:roq_revolt }
					transfer_state_to = event_target:roq_revolt
				}
				transfer_units_fraction = {
					target = event_target:roq_revolt
					stockpile_ratio = 0.05
					army_ratio = 0.05
					navy_ratio = 0.05
					air_ratio = 0.05
				}
			}
			event_target:roq_revolt = {
				declare_war_on = {
					target = ROQ
					type = take_claimed_state
				}
			}
			# Notify all nations at war with ROQ about the splinter
			every_country = {
				limit = {
					has_war_with = ROQ
					NOT = { tag = ROOT }
					NOT = { is_roqualian_splinter = yes }
				}
				country_event = { id = ROQ_event.6 days = 1 }
			}
		}
	}

	option = {
		name = ROQ_event.5.a
	}

	option = {
		name = ROQ_event.5.b
		every_owned_state = {
			limit = { is_core_of = event_target:roq_revolt }
			transfer_state_to = event_target:roq_revolt
		}
		white_peace = event_target:roq_revolt
		trigger = {
			is_ai = no
		}
	}
}

# Notification to nations at war with ROQ: splinter revolt
country_event = {
	id = ROQ_event.6
	title = ROQ_event.6.t
	desc = ROQ_event.6.d
	picture = GFX_report_event_dofus

	is_triggered_only = yes

	# Declare war on the splinter
	option = {
		name = ROQ_event.6.a
		declare_war_on = {
			target = event_target:roq_revolt
			type = take_claimed_state
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				OR = {
					has_fascist_gov_group = yes
					has_neutrality_gov_group = yes
				}
			}
		}
	}

	# Leave them be
	option = {
		name = ROQ_event.6.b
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				OR = {
					has_fascist_gov_group = yes
					has_neutrality_gov_group = yes
				}
			}
		}
	}
}
