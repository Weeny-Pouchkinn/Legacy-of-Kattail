add_namespace = ACR_debug
#Win the war for Amphibia
country_event = {
	id = ACR_debug.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event ACR_debug.1"}
	title = ACR_debug.1.t
	desc = ACR_debug.1.d
	picture = GFX_report_event_valewria
	is_triggered_only = yes

	option = { #To war!!
		name = ACR_event.1.a
		KKN = { retire_all_characters = yes }
		FOD = { #Store him in the ZZZ tag
			KTZ_Loris = { set_nationality = ZZZ }
		}
		TEM = { retire_all_characters = yes }
		ACR = { annex_country = { target = KKN } }
		ACR = { annex_country = { target = FOD } }
		ENO = { annex_country = { target = TEM } }
		if = { #Give back Krokatzian lands
			limit = {
				WAC = {
					NOT = { has_war_with = ACR }
					has_communist_gov_group = yes
				}
			}
			ACR = {
				every_owned_state = {
					limit = { is_claimed_by = WAC }
					transfer_state_to = WAC
				}
			}
		}
		ENO = {
			every_owned_state = {
				limit = { is_claimed_by = ACR }
				transfer_state_to = ACR
			}
		}
		#Reconstruction Zone
		1036 = {
			add_core_of = KKN
		}
		93 = {
			add_core_of = KKN
		}
		ACR = { #Remove faction
			every_other_country = {
				limit = { is_in_faction_with = ACR NOT = { is_puppet_of = ACR } }
				leave_faction = yes
			}
			set_faction_name = "Socintern"
			activate_mission = ACR_complete_reconstruction
			every_owned_state = {
				limit = { is_core_of = KKN }
				transfer_state_to = KKN
			}
			ACR = {
				set_autonomy = {
					target = KKN
					autonomy_state = autonomy_puppet
				}
			}
			KKN = {
				set_cosmetic_tag = ACR_RECONSTRUCTION_ZONE
			}
		}
		news_event = { id = LOK_news.6 days = 2 } 
	}
}
