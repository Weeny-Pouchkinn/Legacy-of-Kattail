#the chase goes wrong and the city is destroyed
country_event = {
	id = TAK_event.9
	immediate = { log = "[GetDateText]: [Root.GetName]: event TAK_event.9"}
	title = TAK_event.9.t
	desc = TAK_event.9.d
	is_triggered_only = yes
	picture = GFX_report_event_kattail

	option = {
		name = TAK_event.9.a
		country_event = { id = TAK_event.10 days = 2 }
		launch_nuke = {
			province = 8983
			use_nuke = no
		}
		set_province_name = { id = 8983 name = "Ruins of Charles" }
		349 = {
			set_building_level = { type = dockyard level = 0 }
			set_building_level = { type = infrastructure level = 0 }
			set_building_level = { type = arms_factory level = 0 }
			set_building_level = { type = industrial_complex level = 0 }
			set_state_category = wasteland
		}
		set_temp_variable = { TAK_population_flees = 349.state_population }
		divide_temp_variable = { TAK_population_flees = 2 }
		347 = { add_manpower = TAK.TAK_population_flees }
		340 = { add_manpower = TAK.TAK_population_flees }
		hidden_effect = { 349 = { add_manpower = -2000000 } }
		set_politics = { ruling_party = democratic }
		add_popularity = {
			ideology = democratic
			popularity = 0.50
		}
	}
}
