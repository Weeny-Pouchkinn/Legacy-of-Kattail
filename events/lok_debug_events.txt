country_event = {
	id = lok_debug.8
	title = ""
	desc = "debug_8"
	is_triggered_only = yes

	immediate = {
		set_variable = { nuke_launch_count = 1 }
		#Pick a nuke type.
		random_list = {
			0 = {
				set_variable = { current_nuke_type = 1 }
				modifier = { add = 1 has_tech = nukes }
				modifier = { factor = 0 has_equipment = { lok_fission_nuke_equipment < 1 } }
			}
			0 = {
				set_variable = { current_nuke_type = 2 }
				modifier = { add = 1 has_tech = fusion_nuke }
				modifier = { factor = 0 has_equipment = { lok_fusion_nuke_equipment < 1 } }
			}
			0 = {
				set_variable = { current_nuke_type = 3 }
				modifier = { add = 1 has_tech = dirty_nuke }
				modifier = { factor = 0 has_equipment = { lok_cobalt_nuke_equipment < 1 } }
			}
			0 = {
				set_variable = { current_nuke_type = 5 }
				modifier = { add = 1 has_tech = ultra_heavy_armor_tech }
				modifier = { factor = 0 has_equipment = { lok_sturmer_nuke_equipment < 1 } }
			}
		}
		random_country = { #PICK A TARGET!!!
			limit = {
				has_war_with = ROOT
				ROOT = { has_country_flag = nuked_by_@PREV }
			}
			random_owned_state = {
				limit = { NOT = { is_core_of = ROOT } }
				save_global_event_target_as = lok_thermonuclear_salvo_state
				ROOT = {
					fire_thermonuclear_salvo = yes #FIRE!!!!
					if = {
						limit = { check_variable = { current_nuke_type = 1 } }
						add_equipment_to_stockpile = {
							type = lok_fission_nuke_equipment
							amount = -1
						}
					}
					else_if = {
						limit = { check_variable = { current_nuke_type = 2 } }
						add_equipment_to_stockpile = {
							type = lok_fusion_nuke_equipment
							amount = -1
						}
					}
					else_if = {
						limit = { check_variable = { current_nuke_type = 3 } }
						add_equipment_to_stockpile = {
							type = lok_cobalt_nuke_equipment
							amount = -1
						}
					}
					else_if = {
						limit = { check_variable = { current_nuke_type = 5 } }
						add_equipment_to_stockpile = {
							type = lok_sturmer_nuke_equipment
							amount = -1
						}
					}
				}
			}
		}
	}

	option = {
		name = "[?ROOT.factories_for_nukes]"
	}
}

country_event = {

	desc = "[?ROOT.nuclear_facility_count]"
	is_triggered_only = yes

	immediate = {
		clear_nuclear_facility_ideas = yes
		set_variable = { nuclear_facility_count = 0 }
		every_controlled_state = {
			limit = { check_variable = { non_damaged_building_level@nuclear_facility > 0 } }
			if = {
				limit = { is_core_of = ROOT }
				add_to_variable = { ROOT.nuclear_facility_count = 2 }
			}
			else = {
				add_to_variable = { ROOT.nuclear_facility_count = 1 }
			}
		}
		if = {
			limit = { check_variable = { nuclear_facility_count > 0 } }
			if = {
				limit = { check_variable = { nuclear_facility_count > 20 } }
				set_variable = { nuclear_facility_count = 20 }
			}
			meta_effect = {
				text = { add_ideas = LOK_nuclear_facility_[NUCLEAR_FACILITY_COUNT] }
				NUCLEAR_FACILITY_COUNT = "[?ROOT.nuclear_facility_count]"
			}
		}
	}

	option = {
		name = "aaah"
	}
}
