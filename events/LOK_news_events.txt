# Battle of Katown (repeatable with ordinal counter)
# Fires when any of the 5 Katown states (509, 34, 36, 44, 45) change hands
# 2-month cooldown between triggers
# ROOT = new controller (attacker), FROM = previous controller (defender), FROM.FROM = state
#########################################
news_event = {
	id = LOK_news.20
	title = "LOK_news.20.t"
	desc = "LOK_news.20.d"
	is_triggered_only = yes
	picture = GFX_news_event_burning_city
	major = yes

	immediate = {
		if = {
			limit = { NOT = { has_global_flag = katown_battle_initialized } }
			set_global_flag = katown_battle_initialized
			set_variable = { global.katown_battle_count = 0 }
		}
		add_to_variable = { global.katown_battle_count = 1 }
	}

	option = {
		name = "LOK_news.20.a"
		trigger = { has_government = communism }
	}
	option = {
		name = "LOK_news.20.b"
		trigger = { has_government = fascism }
