# Event 1: War Initialization (hidden, triggered from NEK_seize_akiba)
country_event = {
	id = KUS_NEK_war.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.1 - KUS-NEK War Initialization"

		# Set global war flag
		set_global_flag = KUS_NEK_war_active

		# Create volunteer division templates for both sides
		KUS = {
			division_template = {
				name = "International Volunteer"
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 0 y = 3 }
				}
			}

			# Set up KUS international support modifier
			set_variable = { KUS_international_support_attack = 0.05 }
			set_variable = { KUS_international_support_defense = 0.05 }
			add_dynamic_modifier = { modifier = KUS_international_support }
		}

		NEK = {
			division_template = {
				name = "International Volunteer"
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 0 y = 3 }
				}
			}

			# Set up NEK international support modifier
			set_variable = { NEK_international_support_attack = 0.05 }
			set_variable = { NEK_international_support_defense = 0.05 }
			add_dynamic_modifier = { modifier = NEK_international_support }
		}

		# Initialize global support tallies
		set_variable = { global.KUS_NEK_tally_guns_kus = 0 }
		set_variable = { global.KUS_NEK_tally_guns_nek = 0 }
		set_variable = { global.KUS_NEK_tally_divs_kus = 0 }
		set_variable = { global.KUS_NEK_tally_divs_nek = 0 }
		set_variable = { global.KUS_NEK_tally_ceasefire_days = 0 }
		set_variable = { global.KUS_NEK_tally_attack_kus = 0 }
		set_variable = { global.KUS_NEK_tally_attack_nek = 0 }
		set_variable = { global.KUS_NEK_tally_defense_kus = 0 }
		set_variable = { global.KUS_NEK_tally_mission_add = 0 }

		# Build supporter array - only majors + SOB + SLR
		every_country = {
			limit = {
				OR = {
					is_major = yes
					tag = SOB
					tag = SLR
				}
				NOT = { tag = KUS }
				NOT = { tag = NEK }
				exists = yes
			}
			add_to_array = { global.KUS_NEK_war_supporters = THIS }

			# Set initial support levels
			if = {
				limit = { tag = ROQ }
				set_variable = { KUS_NEK_support_level = 2 }
			}
			else_if = {
				limit = { has_government = communism }
				set_variable = { KUS_NEK_support_level = -2 }
			}
			else_if = {
				limit = {
					has_government = democratic
					is_major = yes
				}
				set_variable = { KUS_NEK_support_level = 2 }
			}
			else = {
				set_variable = { KUS_NEK_support_level = 0 }
			}
		}
	}
}

# Event 4: Support Choice for Supporting Nations (fired from decision)
country_event = {
	id = KUS_NEK_war.4
	immediate = { log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.4" }
	title = KUS_NEK_war.4.t
	desc = KUS_NEK_war.4.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	# Option A: Support Nekaston
	option = {
		name = KUS_NEK_war.4.a

		# Pay costs immediately
		if = {
			limit = { has_government = communism }
			add_manpower = -16000
			custom_effect_tooltip = KUS_NEK_war_spawn_divs_nek_tt
		}
		if = {
			limit = { NOT = { has_government = communism } }
			send_equipment = {
				target = NEK
				type = infantry_equipment
				amount = 1000
			}
		}

		# Save support choice - will be tallied at end of quarter
		hidden_effect = {
			set_country_flag = KUS_NEK_war_chose_support_nek
			clr_country_flag = KUS_NEK_war_chose_support_kus
			clr_country_flag = KUS_NEK_war_chose_ceasefire
			set_country_flag = { flag = KUS_NEK_war_supported_recently days = 90 value = 1 }

			# Add to tallies
			if = {
				limit = { has_government = communism }
				add_to_variable = { global.KUS_NEK_tally_divs_nek = 2 }
			}
			if = {
				limit = { NOT = { has_government = communism } }
				add_to_variable = { global.KUS_NEK_tally_guns_nek = 1000 }
			}
			add_to_variable = { global.KUS_NEK_tally_attack_nek = 1 }
			add_to_variable = { global.KUS_NEK_tally_mission_add = 5 }
		}

		ai_chance = {
			base = 0
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = -1 } }
			modifier = { add = 100 check_variable = { KUS_NEK_support_level = -2 } }
		}
	}

	# Option B: Support Kusnurk
	option = {
		name = KUS_NEK_war.4.b

		# Pay costs immediately
		if = {
			limit = { has_government = democratic }
			add_manpower = -16000
			custom_effect_tooltip = KUS_NEK_war_spawn_divs_kus_tt
		}
		if = {
			limit = { NOT = { has_government = democratic } }
			send_equipment = {
				target = KUS
				type = infantry_equipment
				amount = 1000
			}
		}

		# Save support choice
		hidden_effect = {
			set_country_flag = KUS_NEK_war_chose_support_kus
			clr_country_flag = KUS_NEK_war_chose_support_nek
			clr_country_flag = KUS_NEK_war_chose_ceasefire
			set_country_flag = { flag = KUS_NEK_war_supported_recently days = 90 value = 1 }

			# Add to tallies
			if = {
				limit = { has_government = democratic }
				add_to_variable = { global.KUS_NEK_tally_divs_kus = 2 }
				add_to_variable = { global.KUS_NEK_tally_attack_kus = 1 }
			}
			if = {
				limit = { NOT = { has_government = democratic } }
				add_to_variable = { global.KUS_NEK_tally_guns_kus = 1000 }
			}
			add_to_variable = { global.KUS_NEK_tally_defense_kus = 1 }

			# If push for neutering is active, KUS supporters add 5 days to mission
			if = {
				limit = { has_global_flag = KUS_NEK_push_for_neutering }
				add_to_variable = { global.KUS_NEK_tally_mission_add = 5 }
			}
		}

		ai_chance = {
			base = 0
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = 1 } }
			modifier = { add = 100 check_variable = { KUS_NEK_support_level = 2 } }
		}
	}

	# Option C: Push for Ceasefire
	option = {
		name = KUS_NEK_war.4.c

		# Save support choice
		hidden_effect = {
			set_country_flag = KUS_NEK_war_chose_ceasefire
			clr_country_flag = KUS_NEK_war_chose_support_kus
			clr_country_flag = KUS_NEK_war_chose_support_nek
			set_country_flag = { flag = KUS_NEK_war_supported_recently days = 90 value = 1 }

			# Add to tallies
			add_to_variable = { global.KUS_NEK_tally_ceasefire_days = 5 }
		}

		ai_chance = {
			base = 0
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = -1 } }
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = 1 } }
			modifier = { add = 100 check_variable = { KUS_NEK_support_level = 0 } }
		}
	}
}

# Event 5: Neuter Nekaston Peace Proposal (sent to NEK from KUS)
country_event = {
	id = KUS_NEK_war.5
	immediate = { log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.5" }
	title = KUS_NEK_war.5.t
	desc = KUS_NEK_war.5.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	# Accept
	option = {
		name = KUS_NEK_war.5.a

		white_peace = KUS

		241 = { transfer_state_to = KUS }
		233 = { transfer_state_to = KUS }

		# DMZ states - dismantle rocket sites and military factories
		248 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}
		746 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}
		240 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}
		287 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}

		add_stability = -0.20
		add_war_support = -0.30

		# War cleanup
		hidden_effect = {
			clr_global_flag = KUS_NEK_war_active
			KUS = {
				if = {
					limit = { has_dynamic_modifier = { modifier = KUS_international_support } }
					remove_dynamic_modifier = { modifier = KUS_international_support }
				}
			}
			if = {
				limit = { has_dynamic_modifier = { modifier = NEK_international_support } }
				remove_dynamic_modifier = { modifier = NEK_international_support }
			}
		}

		KUS = { country_event = LOK_proposal.1 }

		ai_chance = { base = 1000 }
	}

	# Reject (player only)
	option = {
		name = KUS_NEK_war.5.b
		add_stability = -0.05
		add_war_support = 0.10
		KUS = { country_event = LOK_proposal.2 }

		ai_chance = { base = 0 }
	}
}

# Event 30: Quarterly Support Summary - KUS receives
# Description is dynamically built showing what KUS and NEK each received
country_event = {
	id = KUS_NEK_war.30
	immediate = { log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.30" }
	title = KUS_NEK_war.30.t
	desc = KUS_NEK_war.30.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	option = {
		name = KUS_NEK_war.30.a

		# Apply KUS bonuses
		if = {
			limit = { check_variable = { global.KUS_NEK_tally_defense_kus > 0 } }
			set_temp_variable = { temp_def = global.KUS_NEK_tally_defense_kus }
			multiply_temp_variable = { temp_def = 0.01 }
			add_to_variable = { KUS_international_support_defense = temp_def }
		}
		if = {
			limit = { check_variable = { global.KUS_NEK_tally_attack_kus > 0 } }
			set_temp_variable = { temp_atk = global.KUS_NEK_tally_attack_kus }
			multiply_temp_variable = { temp_atk = 0.01 }
			add_to_variable = { KUS_international_support_attack = temp_atk }
		}

		# Spawn volunteer divisions for KUS
		if = {
			limit = { check_variable = { global.KUS_NEK_tally_divs_kus > 0 } }
			capital_scope = {
				create_unit = {
					division = "division_template = \"International Volunteer\" start_experience_factor = 0.1 start_equipment_factor = 1"
					count = global.KUS_NEK_tally_divs_kus
					owner = KUS
				}
			}
		}

		# Ceasefire mission adjustments
		if = {
			limit = { check_variable = { global.KUS_NEK_tally_mission_add > 0 } }
			add_days_mission_timeout = {
				mission = KUS_great_powers_mediate_ceasefire
				days = global.KUS_NEK_tally_mission_add
			}
		}
		if = {
			limit = { check_variable = { global.KUS_NEK_tally_ceasefire_days > 0 } }
			set_temp_variable = { neg_days = global.KUS_NEK_tally_ceasefire_days }
			multiply_temp_variable = { neg_days = -1 }
			add_days_mission_timeout = {
				mission = KUS_great_powers_mediate_ceasefire
				days = neg_days
			}
		}
	}
}
