# Event 1: War Initialization (hidden, triggered from NEK_seize_akiba)
country_event = {
	id = KUS_NEK_war.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.1 - KUS-NEK War Initialization"

		# Set global war flag
		set_global_flag = KUS_NEK_war_active

		# Create volunteer division templates for both sides
		KUS = {
			division_template = {
				name = "International Volunteer"
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 0 y = 3 }
				}
			}

			# Set up KUS international support modifier
			set_variable = { KUS_international_support_attack = 0.05 }
			set_variable = { KUS_international_support_defense = 0.05 }
			add_dynamic_modifier = { modifier = KUS_international_support }
		}

		NEK = {
			division_template = {
				name = "International Volunteer"
				regiments = {
					infantry = { x = 0 y = 0 }
					infantry = { x = 0 y = 1 }
					infantry = { x = 0 y = 2 }
					infantry = { x = 0 y = 3 }
				}
			}

			# Set up NEK international support modifier
			set_variable = { NEK_international_support_attack = 0.05 }
			set_variable = { NEK_international_support_defense = 0.05 }
			add_dynamic_modifier = { modifier = NEK_international_support }
		}

		# Build supporter array
		every_country = {
			limit = {
				OR = {
					is_major = yes
					has_government = communism
				}
				NOT = { tag = KUS }
				NOT = { tag = NEK }
				exists = yes
			}
			add_to_array = { global.KUS_NEK_war_supporters = THIS }

			# Set initial support levels
			if = {
				limit = { tag = ROQ }
				set_variable = { KUS_NEK_support_level = 2 }
			}
			else_if = {
				limit = { has_government = communism }
				set_variable = { KUS_NEK_support_level = -2 }
			}
			else_if = {
				limit = {
					has_government = democratic
					is_major = yes
				}
				set_variable = { KUS_NEK_support_level = 2 }
			}
			else = {
				set_variable = { KUS_NEK_support_level = 0 }
			}
		}
	}
}

# Event 4: Support Choice for Supporting Nations
country_event = {
	id = KUS_NEK_war.4
	immediate = { log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.4" }
	title = KUS_NEK_war.4.t
	desc = KUS_NEK_war.4.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	# Option A: Support Nekaston
	option = {
		name = KUS_NEK_war.4.a

		# Supporter costs
		if = {
			limit = { has_government = communism }
			add_manpower = -16000
			custom_effect_tooltip = KUS_NEK_war_spawn_divs_nek_tt
		}
		if = {
			limit = { NOT = { has_government = communism } }
			send_equipment = {
				target = NEK
				type = infantry_equipment
				amount = 1000
			}
		}

		set_country_flag = { flag = KUS_NEK_war_supported_recently days = 90 value = 1 }

		# Fire notification events
		hidden_effect = {
			NEK = { country_event = { id = KUS_NEK_war.12 } }
			KUS = { country_event = { id = KUS_NEK_war.13 } }
		}

		ai_chance = {
			base = 0
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = -1 } }
			modifier = { add = 100 check_variable = { KUS_NEK_support_level = -2 } }
		}
	}

	# Option B: Support Kusnurk
	option = {
		name = KUS_NEK_war.4.b

		# Supporter costs
		if = {
			limit = { has_government = democratic }
			add_manpower = -16000
			custom_effect_tooltip = KUS_NEK_war_spawn_divs_kus_tt
		}
		if = {
			limit = { NOT = { has_government = democratic } }
			send_equipment = {
				target = KUS
				type = infantry_equipment
				amount = 1000
			}
		}

		set_country_flag = { flag = KUS_NEK_war_supported_recently days = 90 value = 1 }

		# Fire notification events
		hidden_effect = {
			KUS = { country_event = { id = KUS_NEK_war.10 } }
			NEK = { country_event = { id = KUS_NEK_war.11 } }
		}

		ai_chance = {
			base = 0
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = 1 } }
			modifier = { add = 100 check_variable = { KUS_NEK_support_level = 2 } }
		}
	}

	# Option C: Push for Ceasefire
	option = {
		name = KUS_NEK_war.4.c

		set_country_flag = { flag = KUS_NEK_war_supported_recently days = 90 value = 1 }

		# Fire notification events
		hidden_effect = {
			KUS = { country_event = { id = KUS_NEK_war.14 } }
			NEK = { country_event = { id = KUS_NEK_war.15 } }
		}

		ai_chance = {
			base = 0
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = -1 } }
			modifier = { add = 50 check_variable = { KUS_NEK_support_level = 1 } }
			modifier = { add = 100 check_variable = { KUS_NEK_support_level = 0 } }
		}
	}
}

# Event 5: Neuter Nekaston Peace Proposal (sent to NEK from KUS)
country_event = {
	id = KUS_NEK_war.5
	immediate = { log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.5" }
	title = KUS_NEK_war.5.t
	desc = KUS_NEK_war.5.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	# Accept
	option = {
		name = KUS_NEK_war.5.a

		white_peace = KUS

		241 = { transfer_state_to = KUS }
		233 = { transfer_state_to = KUS }

		# DMZ states - dismantle rocket sites and military factories
		248 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}
		746 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}
		240 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}
		287 = {
			set_demilitarized_zone = yes
			remove_building = { type = rocket_site level = 20 }
			remove_building = { type = arms_factory level = 20 }
		}

		add_stability = -0.20
		add_war_support = -0.30

		# War cleanup
		hidden_effect = {
			clr_global_flag = KUS_NEK_war_active
			KUS = {
				if = {
					limit = { has_dynamic_modifier = { modifier = KUS_international_support } }
					remove_dynamic_modifier = { modifier = KUS_international_support }
				}
			}
			if = {
				limit = { has_dynamic_modifier = { modifier = NEK_international_support } }
				remove_dynamic_modifier = { modifier = NEK_international_support }
			}
		}

		KUS = { country_event = LOK_proposal.1 }

		ai_chance = { base = 1000 }
	}

	# Reject (player only)
	option = {
		name = KUS_NEK_war.5.b
		add_stability = -0.05
		add_war_support = 0.10
		KUS = { country_event = LOK_proposal.2 }

		ai_chance = { base = 0 }
	}
}

# Event 10: Support arrives for Kusnurk (KUS notification - applies effects)
country_event = {
	id = KUS_NEK_war.10
	immediate = { log = "[GetDateText]: [Root.GetName]: event KUS_NEK_war.10" }
	title = KUS_NEK_war.10.t
	desc = KUS_NEK_war.10.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	option = {
		name = KUS_NEK_war.10.a

		# Defence always increases when supporting KUS
		add_to_variable = { KUS_international_support_defense = 0.01 }
		custom_effect_tooltip = KUS_NEK_war_defense_up_tt

		# If supporter is democratic, also increase attack and spawn divs
		if = {
			limit = { FROM = { has_government = democratic } }
			add_to_variable = { KUS_international_support_attack = 0.01 }
			custom_effect_tooltip = KUS_NEK_war_attack_up_tt
			capital_scope = {
				create_unit = {
					division = "division_template = \"International Volunteer\" start_experience_factor = 0.1 start_equipment_factor = 1"
					count = 2
					owner = KUS
				}
			}
		}

		# If push for neutering is active, KUS supporters add 5 days to mission
		if = {
			limit = { has_global_flag = KUS_NEK_push_for_neutering }
			add_days_mission_timeout = {
				mission = KUS_great_powers_mediate_ceasefire
				days = 5
			}
			custom_effect_tooltip = KUS_NEK_war_mission_add_5_tt
		}
	}
}
