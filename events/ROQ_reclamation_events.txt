add_namespace = roq_reclamation

# Event 1: Ultimatum/Demand sent to third-party owners of target states
country_event = {
	id = roq_reclamation.1
	title = roq_reclamation.1.t
	desc = roq_reclamation.1.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# Logs or other hidden initialization
		}
	}

	# OPTION A: Hand over all states (Non-Democratic)
	option = {
		name = roq_reclamation.1.a
		trigger = {
			NOT = { has_democratic_gov_group = yes }
		}
		custom_effect_tooltip = roq_reclamation_accept_demands_tt
		every_owned_state = {
			limit = {
				OR = {
					state = 236
					state = 254
					state = 255
					state = 730
					state = 242
					state = 244
				}
			}
			transfer_state_to = ROQ
		}
		ROQ = {
			country_event = { id = roq_reclamation.2 }
			if = {
				limit = {
					236 = { is_owned_by = ROQ }
					254 = { is_owned_by = ROQ }
					255 = { is_owned_by = ROQ }
					730 = { is_owned_by = ROQ }
					242 = { is_owned_by = ROQ }
					244 = { is_owned_by = ROQ }
					has_war = no
				}
				ROQ_reclamation_cleanup = yes
			}
		}
	}

	# OPTION B: Refuse and go to war (Non-Democratic)
	option = {
		name = roq_reclamation.1.b
		trigger = {
			NOT = { has_democratic_gov_group = yes }
		}
		custom_effect_tooltip = roq_reclamation_refuse_demands_tt
		ROQ = {
			declare_war_on = { target = ROOT type = annex_everything }
			set_variable = { reclamation_stage = 1 }
			country_event = { id = roq_reclamation.3 }
		}
	}

	# OPTION C: Compromise and hand over bordering FSR states (Democratic)
	option = {
		name = roq_reclamation.1.c
		trigger = {
			has_democratic_gov_group = yes
		}
		custom_effect_tooltip = roq_reclamation_accept_compromise_tt
		every_owned_state = {
			limit = {
				is_core_of = FSR
				any_neighbor_state = { is_owned_by = ROQ }
			}
			transfer_state_to = ROQ
		}
		ROQ = {
			country_event = { id = roq_reclamation.4 }
			if = {
				limit = {
					236 = { is_owned_by = ROQ }
					254 = { is_owned_by = ROQ }
					255 = { is_owned_by = ROQ }
					730 = { is_owned_by = ROQ }
					242 = { is_owned_by = ROQ }
					244 = { is_owned_by = ROQ }
					has_war = no
				}
				ROQ_reclamation_cleanup = yes
			}
		}
	}

	# OPTION D: Refuse compromise and go to war (Democratic)
	option = {
		name = roq_reclamation.1.e
		trigger = {
			has_democratic_gov_group = yes
		}
		custom_effect_tooltip = roq_reclamation_refuse_compromise_tt
		ROQ = {
			declare_war_on = { target = ROOT type = annex_everything }
			set_variable = { reclamation_stage = 1 }
			country_event = { id = roq_reclamation.5 }
		}
	}
}

# Event 10: Stage 1 completion check event for ROQ (controls all states)
country_event = {
	id = roq_reclamation.10
	title = roq_reclamation.10.t
	desc = roq_reclamation.10.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# Process capture check
		}
	}

	# Option A: Above 75% WS -> Coup & Stage 2
	option = {
		name = roq_reclamation.10.a
		trigger = {
			has_war_support > 0.50
		}
		custom_effect_tooltip = roq_reclamation_escalate_ws_high_tt
		ROQ_reclamation_build_stage_2_targets = yes
		set_variable = { reclamation_stage = 2 }
		add_popularity = {
			ideology = neutrality
			popularity = 0.50
		}
		set_politics = {
			ruling_party = neutrality
			elections_allowed = no
		}
		news_event = { id = LOK_news.14 days = 1 }
	}

	# Option B: Below 75% WS -> Sue for peace and trigger 7-day escalation check
	option = {
		name = roq_reclamation.10.b
		trigger = {
			NOT = { has_war_support > 0.75 }
		}
		custom_effect_tooltip = roq_reclamation_sue_peace_ws_low_tt
		every_country = {
			limit = {
				has_war_with = ROQ
				NOT = { tag = FSR }
				NOT = { tag = DOM }
				OR = {
					AND = {
						is_faction_leader = yes
						is_subject = no
					}
					AND = {
						is_in_faction = no
						is_subject = no
					}
				}
			}
			country_event = { id = roq_reclamation.100 }
		}
		ROQ = {
			country_event = { id = roq_reclamation.110 days = 7 }
		}
	}
}

# Event 100: Peace offer sent to ROQ's enemies (faction leaders and independent nations)
country_event = {
	id = roq_reclamation.100
	title = roq_reclamation.100.t
	desc = roq_reclamation.100.d
	picture = GFX_report_event_generic_diplomacy
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# Empty
		}
	}

	# Option A: Agree to white peace (AI base = 100)
	option = {
		name = roq_reclamation.100.a
		ai_will_do = {
			base = 100
		}
		custom_effect_tooltip = roq_reclamation_accept_peace_tt
		if = {
			limit = {
				FSR = { is_in_faction_with = ROOT }
			}
			FSR = { leave_faction = yes }
		}
		if = {
			limit = {
				DOM = { is_in_faction_with = ROOT }
			}
			DOM = { leave_faction = yes }
		}
		white_peace = ROQ
		every_country = {
			limit = {
				is_in_faction_with = ROOT
				has_war_with = ROQ
			}
			white_peace = ROQ
		}
		every_owned_state = {
			limit = {
				OR = {
					state = 236
					state = 254
					state = 255
					state = 730
					state = 242
					state = 244
				}
			}
			transfer_state_to = ROQ
		}
	}

	# Option B: Refuse peace (AI base = 0)
	option = {
		name = roq_reclamation.100.b
		ai_will_do = {
			base = 0
		}
		custom_effect_tooltip = roq_reclamation_refuse_peace_tt
		# No immediate effect, the ROQ 7-day escalation check will handle this if they are still at war.
	}
}

# Event 110: 7-day post-peace check processor (hidden)
country_event = {
	id = roq_reclamation.110
	hidden = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					any_country = {
						has_war_with = ROQ
						NOT = { tag = FSR }
						NOT = { tag = DOM }
					}
				}
				# Some wars remain -> Refused! Escalate to Stage 2
				ROQ_reclamation_build_stage_2_targets = yes
				set_variable = { reclamation_stage = 2 }
				add_war_support = 0.20
				add_timed_idea = { idea = ROQ_war_exhaustion days = 180 }
				add_popularity = {
					ideology = neutrality
					popularity = 0.50
				}
				set_politics = {
					ruling_party = neutrality
					elections_allowed = no
				}
				news_event = { id = LOK_news.14 days = 1 }
				country_event = { id = roq_reclamation.111 }
			}
			else = {
				# No remaining wars -> Peace succeeded!
				ROQ_reclamation_cleanup = yes
				country_event = { id = roq_reclamation.112 }
			}
		}
	}
}
