# Event 1: Hidden initialization - fires 1 day after ROQ_begin_the_reclamation completes.
# Builds the SSU supporters array, the targets array, initialises modifier variables
# and dynamic modifiers for FSR/DOM, and fires the global flag.
country_event = {
	id = ROQ_reclamation.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event ROQ_reclamation.1 - Reclamation initialization"

		set_global_flag = ROQ_reclamation_active

		# Build the targets array (FSR/DOM who are actually at war with ROQ)
		every_country = {
			limit = {
				OR = { tag = FSR tag = DOM }
				country_exists = THIS
				has_war_with = ROQ
			}
			add_to_array = { global.ROQ_reclamation_targets = THIS }
			add_dynamic_modifier = { modifier = ROQ_reclamation_support }
		}

		# Build the SSU supporters array: MCF + all nations in faction with MCF
		if = {
			limit = { country_exists = MCF }
			MCF = {
				add_to_array = { global.ROQ_reclamation_ssu_supporters = MCF }
				every_country = {
					limit = {
						is_in_faction_with = MCF
						NOT = { tag = MCF }
						country_exists = THIS
					}
					add_to_array = { global.ROQ_reclamation_ssu_supporters = THIS }
				}
			}
		}

		# Initialise support level for each SSU member (0 = Disinterested)
		every_country = {
			limit = {
				is_in_array = { array = global.ROQ_reclamation_ssu_supporters value = THIS }
				country_exists = THIS
			}
			country_event = { id = ROQ_reclamation.2 days = 5 }
			if = {
				limit = { tag = KHA }
				set_variable = { ROQ_reclamation_support_level = 2 }
			}
			else_if = {
				limit = { tag = MCF }
				set_variable = { ROQ_reclamation_support_level = 3 }
			}
			else = {
				set_variable = { ROQ_reclamation_support_level = 1 }
			}
		}

		# Fire Flusiobalkan reaction event for every Flusiobalkan except FSR, DOM and MCF faction members
		every_country = {
			limit = {
				is_flusiobalkan = yes
				NOT = { tag = FSR }
				NOT = { tag = DOM }
				NOT = { is_in_faction_with = MCF }
				country_exists = THIS
			}
			country_event = { id = ROQ_reclamation.30 days = 3 }
		}
	}
}
