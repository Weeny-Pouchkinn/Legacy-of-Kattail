add_namespace = ROQ_reclamation

# Event 1: Hidden initialization - fires 1 day after ROQ_begin_the_reclamation completes.
# Builds the SSU supporters array, the targets array, initialises modifier variables
# and dynamic modifiers for FSR/DOM, and fires the global flag.
country_event = {
	id = ROQ_reclamation.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event ROQ_reclamation.1 - Reclamation initialization"

		set_global_flag = ROQ_reclamation_active

		# Build the targets array (FSR/DOM who are actually at war with ROQ)
		every_country = {
			limit = {
				OR = { tag = FSR tag = DOM }
				country_exists = yes
				has_war_with = ROQ
			}
			add_to_array = { global.ROQ_reclamation_targets = THIS }

			# Initialise dynamic modifier variables
			set_variable = { ROQ_reclamation_air_agility_mod = 0 }
			set_variable = { ROQ_reclamation_air_attack_mod = 0 }
			set_variable = { ROQ_reclamation_air_defence_mod = 0 }
			set_variable = { ROQ_reclamation_army_attack_mod = 0 }
			set_variable = { ROQ_reclamation_army_defence_mod = 0 }
			set_variable = { ROQ_reclamation_breakthrough_mod = 0 }
			set_variable = { ROQ_reclamation_coordination_mod = 0 }
			set_variable = { ROQ_reclamation_supply_consumption_mod = 0 }
			set_variable = { ROQ_reclamation_build_speed_mod = 0 }
			set_variable = { ROQ_reclamation_factory_output_mod = 0 }
			set_variable = { ROQ_reclamation_fort_build_speed_mod = 0 }
			set_variable = { ROQ_reclamation_aa_build_speed_mod = 0 }
			set_variable = { ROQ_reclamation_consumer_goods_mod = 0 }
			add_dynamic_modifier = { modifier = ROQ_reclamation_ssu_support }
		}

		# Build the SSU supporters array: MCF + all nations in faction with MCF
		if = {
			limit = { country_exists = MCF }
			MCF = {
				add_to_array = { global.ROQ_reclamation_ssu_supporters = MCF }
				every_country = {
					limit = {
						is_in_faction_with = MCF
						NOT = { tag = MCF }
						country_exists = yes
					}
					add_to_array = { global.ROQ_reclamation_ssu_supporters = THIS }
				}
			}
		}

		# Initialise support level for each SSU member (0 = Disinterested)
		every_country = {
			limit = {
				is_in_array = { array = global.ROQ_reclamation_ssu_supporters value = THIS }
				country_exists = yes
			}
			set_variable = { ROQ_reclamation_support_level = 0 }
		}
	}
}

# ─────────────────────────────────────────────────────────────────────────────
# Event 50: Hidden cleanup check - fires periodically while system is active,
# cleans up if ROQ is no longer at war with anyone in the targets array.
# Triggered on MCF to avoid running on every country in the world.
# ─────────────────────────────────────────────────────────────────────────────
country_event = {
	id = ROQ_reclamation.50
	hidden = yes
	is_triggered_only = no

	trigger = {
		tag = MCF
		has_global_flag = ROQ_reclamation_active
		NOT = {
			any_country = {
				is_in_array = { array = global.ROQ_reclamation_targets value = THIS }
				country_exists = yes
				has_war_with = ROQ
			}
		}
	}

	mean_time_to_happen = { days = 7 }

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event ROQ_reclamation.50 - Reclamation cleanup (war ended)"
		ROQ_reclamation_cleanup = yes
	}
}
