country_event = {
	id = SPQ_pol_focus.3 # ZETO Membership
	title = "SPQ_pol_focus.3.t"
	desc = "SPQ_pol_focus.3.desc"
	picture = GFX_

	is_triggered_only = yes

	option = {
		name = SPQ_pol_focus.3.a

		every_country = {
			limit = {
				is_in_faction_with = KTN
			}
			give_guarantee = SPQ
			diplomatic_relation = {
				country = SPQ
				relation = non_aggression_pact
			}
			give_market_access = SPQ
		}
		every_country = {
			limit = {
				is_in_faction_with = KTN
			}
			SPQ = {
				give_guarantee = PREV
				diplomatic_relation = {
					country = PREV
					relation = non_aggression_pact
				}
			}
		}

		add_to_variable = {
			SPQ_consumer_goods_factor_var = -0.35
			SPQ_consumer_goods_base = -0.15
		}

		SPQ = {
			set_country_flag = SPQ_in_ZETO
		}
		
		ai_chance = {
			factor = 5
		}
	}
	option = {
		name = SPQ_pol_focus.3.b
		
	}
}
