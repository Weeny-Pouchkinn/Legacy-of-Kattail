add_namespace = PTQ_megacorp

# ==================== DEMAND EVENTS ====================
# Each megacorp periodically demands something from Katzpon
# Triggered from on_monthly in LOK_on_actions.txt

# Pfotenstahl AG demands increased weapons production
country_event = {
	id = PTQ_megacorp.1
	title = PTQ_megacorp.1.t
	desc = PTQ_megacorp.1.d
	picture = GFX_report_event_generic_factory

	is_triggered_only = yes

	trigger = {
		tag = PTQ
		has_country_flag = PTQ_corps_initialized
		NOT = { has_country_flag = PTQ_pfotenstahl_seceded }
	}

	# Comply - Pay equipment
	option = {
		name = PTQ_megacorp.1.a
		ai_chance = { base = 70 }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = -300
		}
		hidden_effect = {
			add_to_variable = { ptq_pfotenstahl_opinion = 10 }
			clamp_variable = { var = ptq_pfotenstahl_opinion min = 0 max = 100 }
			set_country_flag = { flag = PTQ_pfotenstahl_concession_recent days = 45 value = 1 }
			PTQ_recalculate_all_corps = yes
		}
		custom_effect_tooltip = PTQ_pfotenstahl_opinion_increase_tt
	}

	# Compromise - Pay less but less opinion
	option = {
		name = PTQ_megacorp.1.b
		ai_chance = { base = 20 }
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = -100
		}
		hidden_effect = {
			add_to_variable = { ptq_pfotenstahl_opinion = 3 }
			clamp_variable = { var = ptq_pfotenstahl_opinion min = 0 max = 100 }
			PTQ_recalculate_all_corps = yes
		}
		custom_effect_tooltip = PTQ_pfotenstahl_opinion_small_increase_tt
	}

	# Refuse - Opinion penalty + stability hit
	option = {
		name = PTQ_megacorp.1.c
		ai_chance = { base = 10 }
		add_stability = -0.03
		hidden_effect = {
			subtract_from_variable = { ptq_pfotenstahl_opinion = 15 }
			clamp_variable = { var = ptq_pfotenstahl_opinion min = 0 max = 100 }
			PTQ_recalculate_all_corps = yes
			PTQ_check_secession = yes
		}
		custom_effect_tooltip = PTQ_pfotenstahl_opinion_decrease_tt
	}
}

# ==================== SECESSION CRISIS EVENT ====================
# Fires when any corp's opinion drops below 10

country_event = {
	id = PTQ_megacorp.10
	title = PTQ_megacorp.10.t
	desc = PTQ_megacorp.10.d
	picture = GFX_report_event_generic_factory

	is_triggered_only = yes

	# Emergency bailout - Pay heavy PP cost to save the relationship
	option = {
		name = PTQ_megacorp.10.a
		ai_chance = { base = 80 }

		trigger = {
			has_political_power > 149
		}

		add_political_power = -150

		hidden_effect = {
			# Reset the corp's opinion to 25
			if = {
				limit = { has_country_flag = { flag = PTQ_secession_pending_corp value = 1 } }
				set_variable = { ptq_pfotenstahl_opinion = 25 }
			}
			if = {
				limit = { has_country_flag = { flag = PTQ_secession_pending_corp value = 2 } }
				set_variable = { ptq_bergkralle_opinion = 25 }
			}
			if = {
				limit = { has_country_flag = { flag = PTQ_secession_pending_corp value = 3 } }
				set_variable = { ptq_quarzkatz_opinion = 25 }
			}
			if = {
				limit = { has_country_flag = { flag = PTQ_secession_pending_corp value = 4 } }
				set_variable = { ptq_schnurrbart_opinion = 25 }
			}
			if = {
				limit = { has_country_flag = { flag = PTQ_secession_pending_corp value = 5 } }
				set_variable = { ptq_samtpfote_opinion = 25 }
			}
			if = {
				limit = { has_country_flag = { flag = PTQ_secession_pending_corp value = 6 } }
				set_variable = { ptq_nordlicht_opinion = 25 }
			}
			clr_country_flag = PTQ_secession_pending_corp
			# Reset warning flags so they can warn again
			clr_country_flag = PTQ_pfotenstahl_secession_warned
			clr_country_flag = PTQ_bergkralle_secession_warned
			clr_country_flag = PTQ_quarzkatz_secession_warned
			clr_country_flag = PTQ_schnurrbart_secession_warned
			clr_country_flag = PTQ_samtpfote_secession_warned
			clr_country_flag = PTQ_nordlicht_secession_warned
			PTQ_recalculate_all_corps = yes
		}

		custom_effect_tooltip = PTQ_secession_averted_tt
	}

	# Accept the secession - Lose the corp permanently
	option = {
		name = PTQ_megacorp.10.b
		ai_chance = { base = 20 }

		PTQ_execute_secession = yes
		custom_effect_tooltip = PTQ_secession_executed_tt
	}
}
