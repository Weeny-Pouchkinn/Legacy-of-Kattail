	id = KUS_event.50
	title = KUS_event.50.t
	desc = KUS_event.50.desc

	is_triggered_only = yes

	# Accept - join the faction
	option = {
		name = KUS_event.50.a
		KUS = {
			add_to_faction = ROOT
			country_event = { id = KUS_event.51 days = 3 }
		}
		ai_chance = {
			base = 50
			# VIX is very likely to join
			modifier = {
				factor = 10
				tag = VIX
			}
			# Higher opinion = more likely
			modifier = {
				factor = 2
				has_opinion = { target = KUS value > 50 }
			}
			modifier = {
				factor = 1.5
				has_opinion = { target = KUS value > 25 }
			}
			# Weaker nations more likely to join for protection
			modifier = {
				factor = 2
				strength_ratio = { tag = KUS ratio < 0.5 }
			}
			modifier = {
				factor = 1.5
				strength_ratio = { tag = KUS ratio < 0.75 }
			}
		}
	}

	# Refuse
	option = {
		name = KUS_event.50.b
		KUS = {
			country_event = { id = KUS_event.52 days = 3 }
		}
		ai_chance = {
			base = 50
			# VIX almost never refuses
			modifier = {
				factor = 0.05
				tag = VIX
			}
			# Low opinion = more likely to refuse
			modifier = {
				factor = 2
				has_opinion = { target = KUS value < -25 }
			}
			# Stronger nations less interested
			modifier = {
				factor = 2
				strength_ratio = { tag = KUS ratio > 1.5 }
			}
		}
	}
}
