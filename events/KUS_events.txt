# Project Boom - SP Artillery dilemma
country_event = {
	id = KUS_event.1
	title = KUS_event.1.t
	desc = KUS_event.1.desc
	picture = GFX_report_event_generic_factory

	is_triggered_only = yes

	# Option 1: Big Gun Good
	option = {
		name = KUS_event.1.a
		custom_effect_tooltip = KUS_project_boom_gun_tt
		hidden_effect = {
			add_to_variable = { KUS_maavoim_division_attack = 0.10 }
			set_country_flag = KUS_project_boom_gun
		}
	}

	# Option 2: Thick Armor Good
	option = {
		name = KUS_event.1.b
		custom_effect_tooltip = KUS_project_boom_armor_tt
		hidden_effect = {
			add_to_variable = { KUS_maavoim_breakthrough = 0.05 }
			set_country_flag = KUS_project_boom_armor
		}
	}
}

# Aligning the Opposition - triggered by Bolster Presidential Power focus
country_event = {
	id = KUS_event.10
	title = KUS_event.10.t
	desc = KUS_event.10.desc
	picture = GFX_report_event_generic_parliament

	is_triggered_only = yes

	option = {
		name = KUS_event.10.a
		# To Stability!
		if = {
			limit = { has_idea = KUS_underground_extremism }
			remove_ideas = KUS_underground_extremism
		}
		add_stability = 0.05
		add_political_power = 25
	}
}

	title = KUS_event.20.t
	desc = KUS_event.20.desc
	picture = GFX_report_event_generic_riot

	is_triggered_only = yes

	trigger = {
		has_country_flag = KUS_lost_war
		OR = {
			has_idea = KUS_rampant_extremism
			has_idea = KUS_suppressed_extremism
			has_idea = KUS_underground_extremism

		name = KUS_event.20.a

		}
	}

	option = {
		name = KUS_event.20.b
		add_stability = -0.10
		hidden_effect = {
			country_event = { id = KUS_event.21 days = 7 }
		}
	}
}

	id = KUS_event.24
	title = KUS_event.24.t
	desc = KUS_event.24.desc

	is_triggered_only = yes

	option = {
		name = KUS_event.24.a
		set_politics = {
			ruling_party = fascism
			elections_allowed = no
		}
		add_popularity = {
			ideology = fascism
			popularity = 0.30
		}
		if = {
			limit = { has_idea = KUS_rampant_extremism }
			remove_ideas = KUS_rampant_extremism
		}
		if = {
			limit = { has_idea = KUS_suppressed_extremism }
			remove_ideas = KUS_suppressed_extremism
		}
		if = {
			limit = { has_idea = KUS_underground_extremism }
			remove_ideas = KUS_underground_extremism
		}
		if = {
			limit = { has_idea = KUS_technocrat_propaganda }
			remove_ideas = KUS_technocrat_propaganda
		}
		add_ideas = KUS_extremism_ascendant
		create_country_leader = {
			name = "Konstantin Roko"
			picture = "GFX_portrait_KUS_Konstantin_Roko"
			expire = "1965.1.1"
			ideology = fascism_ideology
			traits = { KUS_mad_scientist }
		}
		set_country_flag = KUS_technocratic_coup_complete
		# Demilitarize all states except Jarvalin
		hidden_effect = {
			213 = { set_demilitarized_zone = yes }
			225 = { set_demilitarized_zone = yes }
			229 = { set_demilitarized_zone = yes }
			231 = { set_demilitarized_zone = yes }
			715 = { set_demilitarized_zone = yes }
			716 = { set_demilitarized_zone = yes }
			717 = { set_demilitarized_zone = yes }
		}
	}
}
