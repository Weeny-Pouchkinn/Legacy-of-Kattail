country_event = {
	id = LOK_high_command.1
	immediate = {log = "[GetDateText]: [Root.GetName]: event LOK_high_command.1"}
	title = "LOK_high_command.1.t"
	desc = "LOK_high_command.1.d"
	is_triggered_only = yes

	after = {
		hidden_effect = {
			clr_country_flag = dealing_with_katzen_high_command_member
			event_target:lok_hc_member_being_dealt_with = { clr_character_flag = katzen_high_command_member_not_dealt_with }
			if = {
				limit = { any_character = { has_character_flag = katzen_high_command_member_not_dealt_with } }
				random_character = {
					limit = { has_character_flag = katzen_high_command_member_not_dealt_with }
					save_global_event_target_as = lok_hc_member_being_dealt_with
				}
				set_country_flag = dealing_with_katzen_high_command_member
				country_event = { id = LOK_high_command.1 days = 1 }
			}
		}
	}

	option = { #Recruit!
		name = "LOK_high_command.1.a"
		custom_effect_tooltip = LOK_high_command.1.a.tt
		add_war_support = 0.05
		add_stability = -0.05
		if = {
			limit = { tag = AUR has_completed_focus = AUR_reform_the_high_command }
			set_temp_variable = { herald_sanity_change = 3 }
			AUR_change_herald_sanity = yes
			add_political_power = 50
		}
		event_target:lok_hc_member_being_dealt_with = {
			set_nationality = ROOT
			clear_character_status_flags = yes
		}
		trigger = {
			NOT = {
				AND = {
					tag = AUR
					NOT = { has_completed_focus = AUR_reform_the_high_command }
				}
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 1000
				is_herzlands_warlord = no
				has_government = fascism
			}
		}
	}

	option = { #Let go!
		name = "LOK_high_command.1.b"
		custom_effect_tooltip = LOK_high_command.1.b.tt
		event_target:lok_hc_member_being_dealt_with = {
			clear_character_status_flags = yes
			set_character_flag = leader_is_retired
		}
		trigger = {
			NOT = {
				tag = AUR
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 1000
				is_herzlands_warlord = no
				has_government = neutrality
			}
		}
	}

	option = { #Jail!
		name = "LOK_high_command.1.c"
		custom_effect_tooltip = LOK_high_command.1.c.tt
		add_stability = 0.05
		add_war_support = -0.05
		event_target:lok_hc_member_being_dealt_with = {
			clear_character_status_flags = yes
			set_character_flag = leader_is_jailed
		}
		trigger = {
			NOT = {
				tag = AUR
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 1000
				is_herzlands_warlord = no
				has_government = democratic
			}
		}
	}

	option = { #Kill!
		name = "LOK_high_command.1.e"
		custom_effect_tooltip = LOK_high_command.1.e.tt
		add_political_power = 50
		event_target:lok_hc_member_being_dealt_with = {
			clear_character_status_flags = yes
			set_character_flag = leader_is_dead
			hidden_effect = { retire = yes }
		}
		if = {
			limit = { tag = AUR has_completed_focus = AUR_chew_slowly }
			set_temp_variable = { herald_sanity_change = -5 }
			AUR_change_herald_sanity = yes
		}
		hidden_effect = {
			subtract_from_variable = { katzen_high_command_member_count = 1 }
			subtract_from_variable = { global.katzen_high_command_member_count_total = 1 }
		}
		trigger = {
			NOT = {
				tag = AUR
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 1000
				is_herzlands_warlord = no
				has_government = communism
			}
		}
	}


	option = { #Kill! (REGENT)
		name = "LOK_high_command.1.f"
		add_political_power = 150
		custom_effect_tooltip = LOK_high_command.1.e.tt
		event_target:lok_hc_member_being_dealt_with = {

add_namespace = LOK_state_actions

state_event = { #recolonisation
	id = LOK_state_actions.1
	is_triggered_only = yes
	hidden = yes

	option = {
		save_event_target_as = state_target
		event_target:state_target = {
			subtract_from_variable = { state_actions_recolonize_weeks_left = 1 }
			if = { #If someone else took control of the state just abort this
				limit = { is_controlled_by = AAA }
				if = {
					limit = { check_variable = { state_actions_recolonize_weeks_left = 0 } }
					set_variable = { state_actions_recolonize_button_status = 1 }
					transfer_state_to = ROOT
					set_demilitarized_zone = no
					if = {
						limit = { NOT = { is_core_of = ROOT } }
						add_compliance = 50
					}
				}
				else = {
					state_event = { id = LOK_state_actions.1 days = 7 trigger_for = ROOT }
				}
			}
			else = {
				set_variable = { state_actions_recolonize_button_status = 1 }
			}
		}
	}
}

state_event = { #regent repression
	id = LOK_state_actions.3
	is_triggered_only = yes
	hidden = yes

	option = {
		save_event_target_as = state_target
		event_target:state_target = {
			subtract_from_variable = { state_actions_AUR_repression_weeks_left = 1 }
			if = { #If someone else took control of the state just abort this
				limit = { is_controlled_by = AUR }
				if = {
					limit = { check_variable = { state_actions_AUR_repression_weeks_left = 0 } }
					set_variable = { state_actions_AUR_repression_button_status = 1 }
					add_resistance = -20
					add_resistance_target = -20
					var:controller = { save_current_manpower = yes }
					add_manpower = -10000
					var:controller = { restore_previous_manpower = yes }
					every_neighbor_state = {
						limit = { is_controlled_by = AUR NOT = { is_core_of = AUR } }
						add_resistance = -10
						add_resistance_target = -10
						var:controller = { save_current_manpower = yes }
						add_manpower = -5000
						var:controller = { restore_previous_manpower = yes }
					}
				}
				else = {
					state_event = { id = LOK_state_actions.3 days = 7 trigger_for = ROOT }
				}
			}
			else = {
				set_variable = { state_actions_recolonize_button_status = 1 }
			}
		}
	}
}

		}
	}
}


#Events regarding post-war peacedeals with decisions
add_namespace = lok_peace
country_event = {
	id = lok_peace.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event LOK_peace.1"}
	title = "lok_peace.1.t"
	desc = "lok_peace.1.desc"

	is_triggered_only = yes

	#Release friendly puppet EVERYWHERE
	option = {
		name = "lok_peace.1.a"
		every_owned_state = {
			limit = {
				is_core_of = event_target:peace_puppet_country
				NOT = { is_core_of = ROOT is_claimed_by = ROOT }
			}
			event_target:peace_puppet_country = { transfer_state = PREV }
		}
		puppet = event_target:peace_puppet_country
		if = {
			limit = { NOT = { tag = ROQ } }
			set_autonomy = { target = event_target:peace_puppet_country autonomy_state = autonomy_puppet }
		}
		else_if = {
			limit = { tag = ROQ }
			set_autonomy = { target = event_target:peace_puppet_country autonomy_state = autonomy_roqualian_puppet }
		}
		ai_chance = {
			base = 1
			modifier = { #USSF makes puppets
				factor = 1000
				tag = event_target:ussf_leader
				NOT = { has_country_flag = usf_formed }
			}
			modifier = {
				factor = 0 #If we can just annex more stuff then do that instead
				NOT = { tag = event_target:ussf_leader }
				event_target:peace_puppet_country = {
					any_core_state = {
						OR = {
							is_core_of = ROOT
							is_claimed_by = ROOT
							is_in_array = { ROOT.potential_cores_array = THIS }
							any_neighbor_state = { is_core_of = ROOT }
							any_neighbor_state = { is_claimed_by = ROOT } 
							any_neighbor_state = { is_in_array = { ROOT.potential_cores_array = THIS } }
						}
					}
				}
			}
		}
		hidden_effect = {
			clr_country_flag = deciding_nation_fate
		}
	}

	#Release friendly puppet + annex cores and coreable regions
	option = {
		name = "lok_peace.1.b"
		trigger = {
			event_target:peace_puppet_country = {
				any_core_state = {
					OR = {
						is_core_of = ROOT
						is_claimed_by = ROOT
						is_in_array = { ROOT.potential_cores_array = THIS }
						#Roqualia
						is_in_array = { ROOT.departments_array = THIS } 
					}
				}
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = event_target:peace_puppet_country
				NOT = {
					is_core_of = ROOT
					is_claimed_by = ROOT
					is_in_array = { ROOT.potential_cores_array = THIS } 
				}
			}
			event_target:peace_puppet_country = { transfer_state = PREV }
		}
		if = {
			limit = { country_exists = event_target:peace_puppet_country }
			puppet = event_target:peace_puppet_country
			if = {
				limit = { NOT = { tag = ROQ } }
				set_autonomy = { target = event_target:peace_puppet_country autonomy_state = autonomy_puppet }
			}
			else_if = {
				limit = { tag = ROQ }
				set_autonomy = { target = event_target:peace_puppet_country autonomy_state = autonomy_roqualian_puppet }
			}
		}
		ai_chance = {
			base = 1
			modifier = {
				factor = 1000
				tag = ROQ
				has_government = fascism
			}
			modifier = { #USSF makes puppets
				factor = 0
				tag = event_target:ussf_leader
				NOT = { has_country_flag = usf_formed }
			}
			modifier = { #If we CAN annex more stuff then do it
				factor = 0
				NOT = {
					tag = ROQ
					has_government = fascism
				}
				event_target:peace_puppet_country = {
					OR = {
						any_neighbor_state = { is_core_of = ROOT }
						any_neighbor_state = { is_claimed_by = ROOT } 
						any_neighbor_state = { is_in_array = { ROOT.potential_cores_array = THIS } }
					}
				}
			}
		}
		hidden_effect = {
			clr_country_flag = deciding_nation_fate
		}
	}

	#Release friendly puppet + annex cores and coreable regions, and then go one state beyond that
	option = {
		name = "lok_peace.1.c"
		trigger = {
			event_target:peace_puppet_country = {
				any_core_state = {
					any_neighbor_state = {
						OR = {
							is_core_of = ROOT
							is_claimed_by = ROOT
							is_in_array = { ROOT.potential_cores_array = THIS } 
						}
					}
				}
			}
		}
		every_owned_state = {
			limit = {
				is_core_of = event_target:peace_puppet_country
				NOT = {
					is_core_of = ROOT
					is_claimed_by = ROOT
					is_in_array = { ROOT.potential_cores_array = THIS } 
					any_neighbor_state = { is_core_of = ROOT }
					any_neighbor_state = { is_claimed_by = ROOT } 
					any_neighbor_state = { is_in_array = { ROOT.potential_cores_array = THIS } }
				}
			}
			event_target:peace_puppet_country = { transfer_state = PREV }
		}
		if = {
			limit = { country_exists = event_target:peace_puppet_country }
			puppet = event_target:peace_puppet_country
			if = {
				limit = { NOT = { tag = ROQ } }
				set_autonomy = { target = event_target:peace_puppet_country autonomy_state = autonomy_puppet }
			}
			else_if = {
				limit = { tag = ROQ }
				set_autonomy = { target = event_target:peace_puppet_country autonomy_state = autonomy_roqualian_puppet }
			}
		}
		ai_chance = {
			base = 10
			modifier = { #USSF makes puppets
				factor = 0
				tag = event_target:ussf_leader
				NOT = { has_country_flag = usf_formed }
			}
			modifier = {
 				factor = 0
				tag = ROQ
				has_government = fascism
			}
		}
		hidden_effect = {
			clr_country_flag = deciding_nation_fate
		}
	}
	#Just annex
	option = {
		name = "lok_peace.1.d"
		ai_chance = {
			base = 0
		}
		hidden_effect = {
			clr_country_flag = deciding_nation_fate
		}
	}
}

#herzland warlords mobilize
country_event = {
	id = lok_herzlands.1
	immediate = {log = "[GetDateText]: [Root.GetName]: event lok_herzlands.1"}
	is_triggered_only = yes
	title = lok_herzlands.1.t
	desc = lok_herzlands.1.d
	picture = GFX_report_event_kattail

	option = {
		name = lok_herzlands.1.a
		custom_effect_tooltip = lok_herzlands.1.a.tt
		if = { #Uniters and Hegemon claim it all
			limit = {
				NOT = { tag = AUR }
				OR = {
					has_country_flag = herzlands_warlord_uniter
					has_country_flag = herzlands_warlord_hegemon
				}
			}
			every_state = {
				limit = {
					is_in_array = { global.herzlands_states_array = THIS }
					NOT = { is_core_of = ROOT }
					NOT = { is_claimed_by = ROOT }
					NOT = { is_owned_by = ROOT }
				}
				add_claim_by = ROOT
				reduce_compliance_coring_threshold_by_10 = yes
				reduce_compliance_coring_threshold_by_10 = yes
			}
		}
		else_if = { #Regionalists claim their neighbors
			limit = {
				has_country_flag = herzlands_warlord_regionalist
			}
			every_state = {
				limit = {
					var:owner = { is_neighbor_of = ROOT }
					is_in_array = { global.herzlands_states_array = THIS }
					NOT = { is_core_of = ROOT }
					NOT = { is_claimed_by = ROOT }
					NOT = { is_owned_by = ROOT }
				}
				add_claim_by = ROOT
				reduce_compliance_coring_threshold_by_10 = yes
				reduce_compliance_coring_threshold_by_10 = yes
			}
		}
	}
}
