# KTW informed of food crisis - must secure surrounding states
country_event = {
	id = KTW_event.1
	immediate = { log = "[GetDateText]: [Root.GetName]: event KTW_event.1" }
	title = KTW_event.1.t
	desc = KTW_event.1.d
	picture = GFX_report_event_burning_city

	is_triggered_only = yes

	option = {
		name = KTW_event.1.a
		activate_mission = KTW_secure_periphery
		hidden_effect = {
			set_country_flag = ktw_katown_struggle_active
			# Add cores on required states (they already have them but just in case)
			30 = { add_core_of = KTW }
			27 = { add_core_of = KTW }
			29 = { add_core_of = KTW }
			42 = { add_core_of = KTW }
			50 = { add_core_of = KTW }
			39 = { add_core_of = KTW }
			# Spawn 24 free divisions
			for_loop_effect = {
				end = 24
				509 = {
					create_unit = {
						division = "name = \"Pyramidenwache\" division_template = \"Pyramidenwache\" start_experience_factor = 1.0 start_equipment_factor = 1.0"
						owner = KTW
					}
				}
			}
		}
	}
}

# KTW Katown Struggle Failure - KTW collapses and is partitioned
country_event = {
	id = KTW_event.5
	immediate = { log = "[GetDateText]: [Root.GetName]: event KTW_event.5" }
	title = KTW_event.5.t
	desc = KTW_event.5.d
	picture = GFX_report_event_burning_city

	is_triggered_only = yes

	option = {
		name = KTW_event.5.a
		hidden_effect = {
			clr_country_flag = ktw_katown_struggle_active
			set_country_flag = ktw_lockdown_active

			# White peace everyone
			every_country = {
				limit = { has_war_with = ROOT }
				white_peace = ROOT
			}

			# Distribute states to specific nations if they exist
			if = {
				limit = { country_exists = PAW }
				42 = { if = { limit = { is_owned_by = KTW } transfer_state_to = PAW } }
				50 = { if = { limit = { is_owned_by = KTW } transfer_state_to = PAW } }
				44 = { if = { limit = { is_owned_by = KTW } transfer_state_to = PAW } }
			}
			if = {
				limit = { country_exists = KVK }
				34 = { if = { limit = { is_owned_by = KTW } transfer_state_to = KVK } }
				29 = { if = { limit = { is_owned_by = KTW } transfer_state_to = KVK } }
			}
			if = {
				limit = { country_exists = KKH }
				27 = { if = { limit = { is_owned_by = KTW } transfer_state_to = KKH } }
				30 = { if = { limit = { is_owned_by = KTW } transfer_state_to = KKH } }
				36 = { if = { limit = { is_owned_by = KTW } transfer_state_to = KKH } }
			}
			if = {
				limit = { country_exists = SIL }
				39 = { if = { limit = { is_owned_by = KTW } transfer_state_to = SIL } }
				45 = { if = { limit = { is_owned_by = KTW } transfer_state_to = SIL } }
			}


			for_loop_effect = {
				end = 5
				every_owned_state = {
					limit = { NOT = { state = 509 } }
					random_neighbor_state = {
						limit = { NOT = { is_controlled_by = KTW } }
						var:controller = { save_global_event_target_as = ktw_state_transfer_target }
					}
					transfer_state_to = event_target:ktw_state_transfer_target
				}
			}

			# Add lockdown spirit
			add_ideas = KTW_lockdown_spirit

			# Broadcast collapse news
			news_event = { id = KTW_event.7 hours = 6 }
		}
	}
}
