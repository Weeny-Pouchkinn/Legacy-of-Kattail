scripted_gui = {
    LOK_parliament_launcher_scripted_gui = {
        context_type = player_context
        window_name = "LOK_parliament_launcher_gui"
        parent_window_token = top_bar

        visible = {
            has_country_flag = lok_parliament_initialized
            NOT = { has_country_flag = LOK_parliament_gui_open }
        }

        effects = {
            LOK_parliament_open_button_click = {
                LOK_parliament_gui_open = yes
                LOK_parliament_gui_initialize_action_state = yes
            }
        }
    }

    LOK_parliament_scripted_gui = {
        context_type = player_context
        window_name = "LOK_parliament_gui"
        parent_window_token = top_bar

        visible = { has_country_flag = LOK_parliament_gui_open }
        dirty = lok_parliament_gui_dirty

        dynamic_lists = {
            LOK_parliament_diagram = {
                array = lok_parliament_gui_seat_x
                change_scope = no
                entry_container = LOK_parliament_seat
                index = seat_idx
            }
            LOK_parliament_party_list = {
                array = lok_parliament_gui_party_list
                value = party_idx
                index = party_list_idx
                change_scope = no
                entry_container = LOK_parliament_party_entry
            }
        }

        properties = {
            LOK_parliament_seat_icon = {
                x = lok_parliament_gui_seat_x^seat_idx
                y = lok_parliament_gui_seat_y^seat_idx
                frame = lok_parliament_gui_seat_frame^seat_idx
            }
        }

        effects = {
            LOK_parliament_close_button_click = {
                clr_country_flag = LOK_parliament_gui_open
                add_to_variable = { lok_parliament_gui_dirty = 1 }
            }

            LOK_show_banned_off_button_click  = { LOK_parliament_gui_toggle_banned = yes }
            LOK_show_banned_on_button_click   = { LOK_parliament_gui_toggle_banned = yes }
            LOK_show_zero_off_button_click    = { LOK_parliament_gui_toggle_zero = yes }
            LOK_show_zero_on_button_click     = { LOK_parliament_gui_toggle_zero = yes }
            LOK_sort_seats_button_click       = { LOK_parliament_gui_set_sort_seats = yes }
            LOK_sort_opinion_button_click     = { LOK_parliament_gui_set_sort_opinion = yes }
            LOK_sort_target_button_click      = { LOK_parliament_gui_set_sort_target = yes }
            LOK_sort_popularity_button_click  = { LOK_parliament_gui_set_sort_popularity = yes }
            LOK_sort_orientation_button_click = { LOK_parliament_gui_set_sort_orientation = yes }
            LOK_sort_gov_button_click         = { LOK_parliament_gui_set_sort_gov = yes }

            LOK_party_manage_button_click = { LOK_parliament_gui_toggle_managed_party = yes }

            LOK_party_concessions_button_click      = { LOK_parliament_action_political_concessions = yes }
            LOK_party_bankroll_off_button_click     = { LOK_parliament_action_bankroll_on = yes }
            LOK_party_bankroll_on_button_click      = { LOK_parliament_action_bankroll_off = yes }
            LOK_party_exclude_button_click          = { LOK_parliament_action_exclude_party = yes }
            LOK_party_reincorporate_button_click    = { LOK_parliament_action_reincorporate_party = yes }
            LOK_global_dissolve_button_click        = { LOK_parliament_action_dissolve_assembly = yes }
            LOK_global_emergency_powers_button_click = { LOK_parliament_action_emergency_powers = yes }
        }

        triggers = {
            LOK_show_banned_off_button_visible = { check_variable = { lok_parliament_gui_show_banned = 0 } }
            LOK_show_banned_on_button_visible  = { check_variable = { lok_parliament_gui_show_banned = 1 } }
            LOK_show_zero_off_button_visible   = { check_variable = { lok_parliament_gui_show_zero = 0 } }
            LOK_show_zero_on_button_visible    = { check_variable = { lok_parliament_gui_show_zero = 1 } }

            # Global-management actions.
            LOK_global_dissolve_button_visible = {
                check_variable = { lok_parliament_gui_managed_party < 0 }
                has_idea = lok_legislative_parliament
            }
            LOK_global_emergency_powers_button_visible = {
                check_variable = { lok_parliament_gui_managed_party < 0 }
                NOT = { has_country_flag = lok_has_emergency_powers }
            }

            # Party-management actions.
            LOK_party_concessions_button_visible = {
                check_variable = { lok_parliament_gui_managed_party > -1 }
                check_variable = { lok_parliament_gui_managed_party_banned = 0 }
            }
            LOK_party_concessions_button_enabled = {
                check_variable = { lok_parliament_gui_managed_party_concession_cooldown = 0 }
            }

            LOK_party_bankroll_off_button_visible = {
                check_variable = { lok_parliament_gui_managed_party > -1 }
                NOT = { has_idea = lok_legislative_direct_democracy }
                check_variable = { lok_parliament_gui_managed_party_banned = 0 }
                check_variable = { lok_parliament_gui_managed_party_bankrolled = 0 }
            }
            LOK_party_bankroll_on_button_visible = {
                check_variable = { lok_parliament_gui_managed_party > -1 }
                NOT = { has_idea = lok_legislative_direct_democracy }
                check_variable = { lok_parliament_gui_managed_party_bankrolled = 1 }
            }
            LOK_party_exclude_button_visible = {
                check_variable = { lok_parliament_gui_managed_party > -1 }
                NOT = { has_idea = lok_legislative_direct_democracy }
                check_variable = { lok_parliament_gui_managed_party_banned = 0 }
                check_variable = { lok_parliament_gui_managed_party_ruling = 0 }
            }
            LOK_party_reincorporate_button_visible = {
                check_variable = { lok_parliament_gui_managed_party > -1 }
                NOT = { has_idea = lok_legislative_direct_democracy }
                check_variable = { lok_parliament_gui_managed_party_banned = 1 }
            }
        }
    }
