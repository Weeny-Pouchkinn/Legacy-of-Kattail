# DEBUG ONLY. Remove this file once real state-political initialization exists.
#
# Initializes on a fresh game via on_startup AND, for convenience while developing,
# also on the first daily pulse of an existing save that does not yet have the v2 flag.

on_actions = {
    on_startup = {
        effect = {
            LOK_DEBUG_initialize_state_politics_v2 = yes
        }
    }

    on_daily = {
        effect = {
            if = {
                limit = { NOT = { has_global_flag = LOK_debug_state_politics_v2_initialized } }
                LOK_DEBUG_initialize_state_politics_v2 = yes
            }
        }
    }
}
