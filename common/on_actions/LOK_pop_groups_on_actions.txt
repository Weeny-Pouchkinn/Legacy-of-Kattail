on_actions = {
    # Pop groups themselves are now static HISTORY data generated into each
    # playable country's state file. Startup only derives totals/caches and
    # schedules duplicate cleanup; it does NOT invent demographic data.
    on_startup = {
        effect = {
            every_country = {
                limit = { is_non_playable_country = no }

                every_owned_state = {
                    limit = { check_variable = { lok_pop_amount_array^num > 0 } }
                    LOK_rebuild_pop_caches = yes
                }

                country_event = { id = LOK_pop_groups.1 days = 180 }
            }
        }
    }
}
