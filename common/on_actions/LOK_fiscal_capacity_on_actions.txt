on_actions = {
	# No generic country-created on-action is exposed in the repository's current
	# supported on-action set. The monthly fallback below catches dynamic countries.
	on_monthly = {
		effect = {
			LOK_initialize_fiscal_capacity_country = yes
		}
	}

	on_puppet = {
		effect = {
			LOK_initialize_fiscal_capacity_country = yes
		}
	}

	on_release_as_puppet = {
		effect = {
			LOK_initialize_fiscal_capacity_country = yes
		}
	}

	on_civil_war_end = {
		effect = {
			LOK_initialize_fiscal_capacity_country = yes
		}
	}
}
