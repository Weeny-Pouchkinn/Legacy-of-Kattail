# Pawris Authority - Dynamic Modifiers
# These modifiers use variables so they can be modified by later focuses

# Local Soviet Administration - granted by "Form Regional Soviets"
# Modified by "Form a Syndicalist State" (-20% consumer goods, +0.1 PP)

	enable = { always = yes }
	icon = GFX_idea_PAW_local_soviet_administration

	consumer_goods_factor = PAW_local_soviet_admin_cg
	production_speed_buildings_factor = 0.10
	industrial_capacity_factory = 0.10
	political_power_gain = PAW_local_soviet_admin_pp
}

# Syndicalist State - granted by "Form a Syndicalist State"

	icon = GFX_idea_PAW_syndicalist_state

}
