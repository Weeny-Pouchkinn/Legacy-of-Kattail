#You also have LOK_alert_index which is the current index of LOK_alert_type_array. Use this to access data from other arrays.
LOK_alert_icon_click_effect = {
	if = {
		limit = { 
			check_variable = { LOK_alert_id = 1 }
		}
		var:LOK_alert_location_array^LOK_alert_index = { PREV = { goto_state = PREV } }
	}
	else_if = {
		limit = {
			check_variable = { LOK_alert_id = 2 }
		}
	}
}
