# -----------------------------------------------------------------------------
# PUBLIC: change one species share by a percentage-point amount.
# Caller inputs MUST be temp variables:
#   lok_distribution_species      = species array index (0..60)
#   lok_distribution_share_change = signed share, e.g. 0.05 or -0.10
# -----------------------------------------------------------------------------
LOK_change_state_species_share = {
    resize_array = { array = lok_state_species_array value = 0 size = 61 }
    set_temp_variable = { lok_distribution_actual_change = 0 }

    if = {
        limit = {
            check_variable = { lok_distribution_species > -1 }
            check_variable = { lok_distribution_species < 61 }
        }

        # Find the target value by iterating the actual numeric array.
        set_temp_variable = { lok_dist_old_share = 0 }
        for_each_loop = {
            array = lok_state_species_array
            value = lok_dist_scan_share
            index = lok_dist_scan_i
            if = {
                limit = { check_variable = { lok_dist_scan_i = lok_distribution_species } }
                set_temp_variable = { lok_dist_old_share = lok_dist_scan_share }
            }
        }
        clamp_temp_variable = { var = lok_dist_old_share min = 0 max = 1 }

        set_temp_variable = { lok_dist_new_share = lok_dist_old_share }
        add_to_temp_variable = { lok_dist_new_share = lok_distribution_share_change }
        clamp_temp_variable = { var = lok_dist_new_share min = 0 max = 1 }

        set_temp_variable = { lok_distribution_actual_change = lok_dist_new_share }
        subtract_from_temp_variable = { lok_distribution_actual_change = lok_dist_old_share }

        set_temp_variable = { lok_dist_old_rest = 1 }
        subtract_from_temp_variable = { lok_dist_old_rest = lok_dist_old_share }

        # A negative change from a 100%-homogeneous state has nowhere to go.
        if = {
            limit = {
                check_variable = { lok_dist_old_rest < 0.000001 }
                check_variable = { lok_distribution_actual_change < 0 }
            }
            set_temp_variable = { lok_dist_new_share = lok_dist_old_share }
            set_temp_variable = { lok_distribution_actual_change = 0 }
        }
    }

    custom_effect_tooltip = LOK_change_state_species_share_tt

    hidden_effect = {
        if = {
            limit = {
                check_variable = { lok_distribution_species > -1 }
                check_variable = { lok_distribution_species < 61 }
                check_variable = { lok_dist_old_rest > 0.000001 }
            }

            set_temp_variable = { lok_dist_new_rest = 1 }
            subtract_from_temp_variable = { lok_dist_new_rest = lok_dist_new_share }
            set_temp_variable = { lok_dist_scale = lok_dist_new_rest }
            divide_temp_variable = { lok_dist_scale = lok_dist_old_rest }

            # Scale every non-target slot in-place. Array arithmetic preserves the
            # actual floating-point/fixed-point value instead of rebuilding it via set.
            for_each_loop = {
                array = lok_state_species_array
                value = lok_dist_scan_share
                index = lok_dist_scan_i
                if = {
                    limit = { NOT = { check_variable = { lok_dist_scan_i = lok_distribution_species } } }
                    multiply_variable = { lok_state_species_array^lok_dist_scan_i = lok_dist_scale }
                }
            }

            # Replace target without set_variable-on-array-element.
            subtract_from_variable = { lok_state_species_array^lok_distribution_species = lok_dist_old_share }
            add_to_variable = { lok_state_species_array^lok_distribution_species = lok_dist_new_share }

            LOK_rebuild_state_distribution_caches = yes
        }
    }
}
