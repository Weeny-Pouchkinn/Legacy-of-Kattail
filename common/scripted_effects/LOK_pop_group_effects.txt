# ============================================================================
# LEGACY OF KATTAIL - POPULATION GROUP TOOLKIT
# ============================================================================
# STATE SCOPE unless a comment says otherwise.
#
# AUTHORITATIVE DATA
# ------------------
# A population group is one row spread across five parallel state arrays.
# The same index in all five arrays is the same population group:
#
#   lok_pop_species_array
#   lok_pop_culture_array
#   lok_pop_politics_array
#   lok_pop_amount_array       # thousands of people
#   lok_pop_reserved_array     # fifth field, reserved; always 0 for now
#
# IMPORTANT SCRIPTING RULE
# ------------------------
# LoK does NOT use pseudo-parameter calls such as:
#
#   LOK_add_pop_group = { SPECIES = 1 CULTURE = 5 ... }
#
# Callers must set the documented TEMP variables immediately before calling the
# effect with '= yes'.  Examples are included below.
# ============================================================================


# ----------------------------------------------------------------------------
# LOW LEVEL - delete the row at lok_pop_remove_index.
# ----------------------------------------------------------------------------
LOK_remove_pop_group_at_index = {
    remove_from_array = { array = lok_pop_species_array index = lok_pop_remove_index }
    remove_from_array = { array = lok_pop_culture_array index = lok_pop_remove_index }
    remove_from_array = { array = lok_pop_politics_array index = lok_pop_remove_index }
    remove_from_array = { array = lok_pop_amount_array index = lok_pop_remove_index }
    remove_from_array = { array = lok_pop_reserved_array index = lok_pop_remove_index }
}


# ----------------------------------------------------------------------------
# LOW LEVEL - add the group stored in lok_pop_input_* or merge it into an exact
# existing match.
#
# Expected TEMP variables:
#   lok_pop_input_species
#   lok_pop_input_culture
#   lok_pop_input_politics
#   lok_pop_input_amount
#   lok_pop_input_reserved
# ----------------------------------------------------------------------------
LOK_add_or_merge_pop_group_internal = {
    if = {
        limit = { check_variable = { lok_pop_input_amount > 0 } }

        set_temp_variable = { lok_pop_found_index = -1 }
        set_temp_variable = { lok_pop_find_i = 0 }

        while_loop_effect = {
            limit = {
                check_variable = { lok_pop_find_i < lok_pop_amount_array^num }
                check_variable = { lok_pop_found_index = -1 }
            }

            set_temp_variable = { lok_pop_find_species = lok_pop_species_array^lok_pop_find_i }
            set_temp_variable = { lok_pop_find_culture = lok_pop_culture_array^lok_pop_find_i }
            set_temp_variable = { lok_pop_find_politics = lok_pop_politics_array^lok_pop_find_i }
            set_temp_variable = { lok_pop_find_reserved = lok_pop_reserved_array^lok_pop_find_i }

            if = {
                limit = {
                    check_variable = { lok_pop_find_species = lok_pop_input_species }
                    check_variable = { lok_pop_find_culture = lok_pop_input_culture }
                    check_variable = { lok_pop_find_politics = lok_pop_input_politics }
                    check_variable = { lok_pop_find_reserved = lok_pop_input_reserved }
                }
                set_temp_variable = { lok_pop_found_index = lok_pop_find_i }
            }
            else = {
                add_to_temp_variable = { lok_pop_find_i = 1 }
            }
        }

        if = {
            limit = { check_variable = { lok_pop_found_index > -1 } }
            add_to_variable = { lok_pop_amount_array^lok_pop_found_index = lok_pop_input_amount }
        }
        else = {
            add_to_array = { array = lok_pop_species_array value = lok_pop_input_species }
            add_to_array = { array = lok_pop_culture_array value = lok_pop_input_culture }
            add_to_array = { array = lok_pop_politics_array value = lok_pop_input_politics }
            add_to_array = { array = lok_pop_amount_array value = lok_pop_input_amount }
            add_to_array = { array = lok_pop_reserved_array value = lok_pop_input_reserved }
        }
    }
}


# ----------------------------------------------------------------------------
# PUBLIC - add one exact group, using TEMP variables prepared by the caller.
# Reserved is forced to 0 in this first pass.
#
# Example, in STATE scope:
#
#   set_temp_variable = { lok_pop_input_species = 1 }
#   set_temp_variable = { lok_pop_input_culture = 5 }
#   set_temp_variable = { lok_pop_input_politics = 1 }
#   set_temp_variable = { lok_pop_input_amount = 250 }
#   LOK_add_prepared_pop_group = yes
# ----------------------------------------------------------------------------
LOK_add_prepared_pop_group = {
    set_temp_variable = { lok_pop_input_reserved = 0 }
    LOK_add_or_merge_pop_group_internal = yes
    LOK_prune_empty_pop_groups = yes
    LOK_rebuild_pop_caches = yes
}


# ----------------------------------------------------------------------------
# Remove groups which have effectively reached zero.
# ----------------------------------------------------------------------------
LOK_prune_empty_pop_groups = {
    set_temp_variable = { lok_pop_prune_i = 0 }

    while_loop_effect = {
        limit = { check_variable = { lok_pop_prune_i < lok_pop_amount_array^num } }

        set_temp_variable = { lok_pop_prune_amount = lok_pop_amount_array^lok_pop_prune_i }

        if = {
            limit = { check_variable = { lok_pop_prune_amount < 0.0005 } }
            set_temp_variable = { lok_pop_remove_index = lok_pop_prune_i }
            LOK_remove_pop_group_at_index = yes
            # Do NOT increment. The next row shifted into the same index.
        }
        else = {
            add_to_temp_variable = { lok_pop_prune_i = 1 }
        }
    }
}


# ----------------------------------------------------------------------------
# Merge every identical five-field row in the state.
# ----------------------------------------------------------------------------
LOK_merge_identical_pop_groups = {
    LOK_prune_empty_pop_groups = yes

    set_temp_variable = { lok_pop_merge_i = 0 }

    while_loop_effect = {
        limit = { check_variable = { lok_pop_merge_i < lok_pop_amount_array^num } }

        set_temp_variable = { lok_pop_merge_species = lok_pop_species_array^lok_pop_merge_i }
        set_temp_variable = { lok_pop_merge_culture = lok_pop_culture_array^lok_pop_merge_i }
        set_temp_variable = { lok_pop_merge_politics = lok_pop_politics_array^lok_pop_merge_i }
        set_temp_variable = { lok_pop_merge_reserved = lok_pop_reserved_array^lok_pop_merge_i }

        set_temp_variable = { lok_pop_merge_j = lok_pop_merge_i }
        add_to_temp_variable = { lok_pop_merge_j = 1 }

        while_loop_effect = {
            limit = { check_variable = { lok_pop_merge_j < lok_pop_amount_array^num } }

            set_temp_variable = { lok_pop_merge_j_species = lok_pop_species_array^lok_pop_merge_j }
            set_temp_variable = { lok_pop_merge_j_culture = lok_pop_culture_array^lok_pop_merge_j }
            set_temp_variable = { lok_pop_merge_j_politics = lok_pop_politics_array^lok_pop_merge_j }
            set_temp_variable = { lok_pop_merge_j_reserved = lok_pop_reserved_array^lok_pop_merge_j }

            if = {
                limit = {
                    check_variable = { lok_pop_merge_species = lok_pop_merge_j_species }
                    check_variable = { lok_pop_merge_culture = lok_pop_merge_j_culture }
                    check_variable = { lok_pop_merge_politics = lok_pop_merge_j_politics }
                    check_variable = { lok_pop_merge_reserved = lok_pop_merge_j_reserved }
                }

                set_temp_variable = { lok_pop_merge_j_amount = lok_pop_amount_array^lok_pop_merge_j }
                add_to_variable = { lok_pop_amount_array^lok_pop_merge_i = lok_pop_merge_j_amount }
                set_temp_variable = { lok_pop_remove_index = lok_pop_merge_j }
                LOK_remove_pop_group_at_index = yes
                # Do NOT increment j. The next row shifted into j.
            }
            else = {
                add_to_temp_variable = { lok_pop_merge_j = 1 }
            }
        }

        add_to_temp_variable = { lok_pop_merge_i = 1 }
    }

    LOK_rebuild_pop_caches = yes
}


# ============================================================================
# AGGREGATE TOTALS
# ============================================================================
# These are DERIVED caches. Pop-group arrays above remain authoritative.
#
# Species and cultures use sparse ID/amount pairs because IDs are not compact:
#   lok_state_species_total_ids_array
#   lok_state_species_total_amount_array
#   lok_state_culture_total_ids_array
#   lok_state_culture_total_amount_array
#
# Politics has exactly ten entries, therefore:
#   lok_state_parties_array^0..9
# stores RAW population amounts (in thousands) by ideology.
#
# LOK_rebuild_state_political_pie is then called. That effect normalizes these
# raw totals into the state political pie/map-mode distribution.
# ============================================================================

LOK_add_current_pop_to_species_total = {
    set_temp_variable = { lok_pop_cache_found = -1 }
    set_temp_variable = { lok_pop_cache_scan = 0 }

    while_loop_effect = {
        limit = {
            check_variable = { lok_pop_cache_scan < lok_state_species_total_ids_array^num }
            check_variable = { lok_pop_cache_found = -1 }
        }

        set_temp_variable = { lok_pop_cache_scan_id = lok_state_species_total_ids_array^lok_pop_cache_scan }

        if = {
            limit = { check_variable = { lok_pop_cache_scan_id = lok_pop_cache_species } }
            set_temp_variable = { lok_pop_cache_found = lok_pop_cache_scan }
        }
        else = {
            add_to_temp_variable = { lok_pop_cache_scan = 1 }
        }
    }

    if = {
        limit = { check_variable = { lok_pop_cache_found > -1 } }
        add_to_variable = { lok_state_species_total_amount_array^lok_pop_cache_found = lok_pop_cache_amount }
    }
    else = {
        add_to_array = { array = lok_state_species_total_ids_array value = lok_pop_cache_species }
        add_to_array = { array = lok_state_species_total_amount_array value = lok_pop_cache_amount }
    }
}

LOK_add_current_pop_to_culture_total = {
    set_temp_variable = { lok_pop_cache_found = -1 }
    set_temp_variable = { lok_pop_cache_scan = 0 }

    while_loop_effect = {
        limit = {
            check_variable = { lok_pop_cache_scan < lok_state_culture_total_ids_array^num }
            check_variable = { lok_pop_cache_found = -1 }
        }

        set_temp_variable = { lok_pop_cache_scan_id = lok_state_culture_total_ids_array^lok_pop_cache_scan }

        if = {
            limit = { check_variable = { lok_pop_cache_scan_id = lok_pop_cache_culture } }
            set_temp_variable = { lok_pop_cache_found = lok_pop_cache_scan }
        }
        else = {
            add_to_temp_variable = { lok_pop_cache_scan = 1 }
        }
    }

    if = {
        limit = { check_variable = { lok_pop_cache_found > -1 } }
        add_to_variable = { lok_state_culture_total_amount_array^lok_pop_cache_found = lok_pop_cache_amount }
    }
    else = {
        add_to_array = { array = lok_state_culture_total_ids_array value = lok_pop_cache_culture }
        add_to_array = { array = lok_state_culture_total_amount_array value = lok_pop_cache_amount }
    }
}

LOK_rebuild_pop_caches = {
    set_variable = { lok_pop_total_k = 0 }
    set_variable = { lok_pop_group_count = 0 }

    clear_array = lok_state_species_total_ids_array
    clear_array = lok_state_species_total_amount_array
    clear_array = lok_state_culture_total_ids_array
    clear_array = lok_state_culture_total_amount_array

    clear_array = lok_state_parties_array
    resize_array = { array = lok_state_parties_array value = 0 size = 10 }

    set_temp_variable = { lok_pop_cache_i = 0 }

    while_loop_effect = {
        limit = { check_variable = { lok_pop_cache_i < lok_pop_amount_array^num } }

        set_temp_variable = { lok_pop_cache_amount = lok_pop_amount_array^lok_pop_cache_i }

        if = {
            limit = { check_variable = { lok_pop_cache_amount > 0 } }

            add_to_variable = { lok_pop_total_k = lok_pop_cache_amount }
            add_to_variable = { lok_pop_group_count = 1 }

            set_temp_variable = { lok_pop_cache_species = lok_pop_species_array^lok_pop_cache_i }
            set_temp_variable = { lok_pop_cache_culture = lok_pop_culture_array^lok_pop_cache_i }
            set_temp_variable = { lok_pop_cache_politics = lok_pop_politics_array^lok_pop_cache_i }

            LOK_add_current_pop_to_species_total = yes
            LOK_add_current_pop_to_culture_total = yes

            if = {
                limit = {
                    check_variable = { lok_pop_cache_politics > -1 }
                    check_variable = { lok_pop_cache_politics < 10 }
                }
                add_to_variable = { lok_state_parties_array^lok_pop_cache_politics = lok_pop_cache_amount }
            }
        }

        add_to_temp_variable = { lok_pop_cache_i = 1 }
    }

    # The political pie/map mode is a projection of the real pop groups.
    LOK_rebuild_state_political_pie = yes

    # Species/culture pies use the sparse aggregate arrays built just above.
    # This is event-driven: they are rebuilt only when the authoritative pop groups change.
    LOK_rebuild_state_demographic_pies = yes

    add_to_variable = { lok_pop_gui_dirty = 1 }
}


# ============================================================================
# FILTERED POPULATION CHANGE TOOLKIT
# ============================================================================
# -1 means ANY for each filter.
# Amount is in thousands. Negative removes, positive adds.
# Matching groups are changed proportionally.
# If removal asks for more than exists, all matching pops are removed.
# If addition finds no matching groups, a new group is created ONLY if species,
# culture and politics are all specified rather than -1.
#
# Example: remove one million Katzen Socialists of any culture:
#
#   set_temp_variable = { lok_pop_change_amount = -1000 }
#   set_temp_variable = { lok_pop_filter_species = 1 }
#   set_temp_variable = { lok_pop_filter_culture = -1 }
#   set_temp_variable = { lok_pop_filter_politics = 1 }
#   LOK_change_state_population = yes
# ============================================================================

LOK_test_current_pop_group_filters = {
    set_temp_variable = { lok_pop_matches_current = 1 }

    set_temp_variable = { lok_pop_current_species = lok_pop_species_array^lok_pop_change_i }
    set_temp_variable = { lok_pop_current_culture = lok_pop_culture_array^lok_pop_change_i }
    set_temp_variable = { lok_pop_current_politics = lok_pop_politics_array^lok_pop_change_i }

    if = {
        limit = {
            NOT = { check_variable = { lok_pop_filter_species = -1 } }
            NOT = { check_variable = { lok_pop_current_species = lok_pop_filter_species } }
        }
        set_temp_variable = { lok_pop_matches_current = 0 }
    }

    if = {
        limit = {
            NOT = { check_variable = { lok_pop_filter_culture = -1 } }
            NOT = { check_variable = { lok_pop_current_culture = lok_pop_filter_culture } }
        }
        set_temp_variable = { lok_pop_matches_current = 0 }
    }

    if = {
        limit = {
            NOT = { check_variable = { lok_pop_filter_politics = -1 } }
            NOT = { check_variable = { lok_pop_current_politics = lok_pop_filter_politics } }
        }
        set_temp_variable = { lok_pop_matches_current = 0 }
    }
}

LOK_change_state_population = {
    # Caller's temp variables are intentionally used directly. Do not convert
    # this back into a pseudo-parameterized scripted effect.
    custom_effect_tooltip = LOK_change_state_population_tt

    hidden_effect = {
        LOK_change_state_population_internal = yes
    }
}

LOK_change_state_population_internal = {
    set_temp_variable = { lok_pop_matching_total = 0 }
    set_temp_variable = { lok_pop_actual_change = 0 }
    set_temp_variable = { lok_pop_change_i = 0 }

    # Pass 1: sum matching population.
    while_loop_effect = {
        limit = { check_variable = { lok_pop_change_i < lok_pop_amount_array^num } }

        LOK_test_current_pop_group_filters = yes

        if = {
            limit = { check_variable = { lok_pop_matches_current = 1 } }

            set_temp_variable = { lok_pop_current_amount = lok_pop_amount_array^lok_pop_change_i }

            if = {
                limit = { check_variable = { lok_pop_current_amount > 0 } }
                add_to_temp_variable = { lok_pop_matching_total = lok_pop_current_amount }
            }
        }

        add_to_temp_variable = { lok_pop_change_i = 1 }
    }

    # Removal.
    if = {
        limit = {
            check_variable = { lok_pop_change_amount < 0 }
            check_variable = { lok_pop_matching_total > 0 }
        }

        set_temp_variable = { lok_pop_requested_loss = lok_pop_change_amount }
        multiply_temp_variable = { lok_pop_requested_loss = -1 }

        if = {
            limit = { check_variable = { lok_pop_requested_loss > lok_pop_matching_total } }
            set_temp_variable = { lok_pop_requested_loss = lok_pop_matching_total }
        }

        set_temp_variable = { lok_pop_actual_change = lok_pop_requested_loss }
        multiply_temp_variable = { lok_pop_actual_change = -1 }

        set_temp_variable = { lok_pop_change_i = 0 }

        while_loop_effect = {
            limit = { check_variable = { lok_pop_change_i < lok_pop_amount_array^num } }

            LOK_test_current_pop_group_filters = yes

            if = {
                limit = { check_variable = { lok_pop_matches_current = 1 } }

                set_temp_variable = { lok_pop_current_amount = lok_pop_amount_array^lok_pop_change_i }
                set_temp_variable = { lok_pop_group_delta = lok_pop_current_amount }
                divide_temp_variable = { lok_pop_group_delta = lok_pop_matching_total }
                multiply_temp_variable = { lok_pop_group_delta = lok_pop_requested_loss }
                subtract_from_variable = { lok_pop_amount_array^lok_pop_change_i = lok_pop_group_delta }
            }

            add_to_temp_variable = { lok_pop_change_i = 1 }
        }
    }

    # Addition.
    else_if = {
        limit = { check_variable = { lok_pop_change_amount > 0 } }

        if = {
            limit = { check_variable = { lok_pop_matching_total > 0 } }

            set_temp_variable = { lok_pop_actual_change = lok_pop_change_amount }
            set_temp_variable = { lok_pop_change_i = 0 }

            while_loop_effect = {
                limit = { check_variable = { lok_pop_change_i < lok_pop_amount_array^num } }

                LOK_test_current_pop_group_filters = yes

                if = {
                    limit = { check_variable = { lok_pop_matches_current = 1 } }

                    set_temp_variable = { lok_pop_current_amount = lok_pop_amount_array^lok_pop_change_i }
                    set_temp_variable = { lok_pop_group_delta = lok_pop_current_amount }
                    divide_temp_variable = { lok_pop_group_delta = lok_pop_matching_total }
                    multiply_temp_variable = { lok_pop_group_delta = lok_pop_change_amount }
                    add_to_variable = { lok_pop_amount_array^lok_pop_change_i = lok_pop_group_delta }
                }

                add_to_temp_variable = { lok_pop_change_i = 1 }
            }
        }
        else_if = {
            limit = {
                check_variable = { lok_pop_filter_species > -1 }
                check_variable = { lok_pop_filter_culture > -1 }
                check_variable = { lok_pop_filter_politics > -1 }
            }

            set_temp_variable = { lok_pop_input_species = lok_pop_filter_species }
            set_temp_variable = { lok_pop_input_culture = lok_pop_filter_culture }
            set_temp_variable = { lok_pop_input_politics = lok_pop_filter_politics }
            set_temp_variable = { lok_pop_input_amount = lok_pop_change_amount }
            set_temp_variable = { lok_pop_input_reserved = 0 }

            LOK_add_or_merge_pop_group_internal = yes
            set_temp_variable = { lok_pop_actual_change = lok_pop_change_amount }
        }
    }

    # Keep vanilla state population synchronized with the simulated groups.
    if = {
        limit = { NOT = { check_variable = { lok_pop_actual_change = 0 } } }

        set_temp_variable = { lok_pop_actual_change_people = lok_pop_actual_change }
        multiply_temp_variable = { lok_pop_actual_change_people = 1000 }
        round_temp_variable = lok_pop_actual_change_people
        add_manpower = lok_pop_actual_change_people
    }

    LOK_prune_empty_pop_groups = yes
    LOK_rebuild_pop_caches = yes
}
