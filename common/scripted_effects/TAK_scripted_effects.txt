TAK_update_painting_stats = {
	TAK_calculate_painting_power = yes
	TAK_calculate_painting_output = yes
}

TAK_calculate_painting_power = {
	set_variable = { TAK_painting_power = 0 }
	#Iterate through the painter array and sum up the highest values equal to the brush count
	#First we copy the whole painter array to a temp array we can mess with
	clear_array = TAK_painter_list_temp
	for_each_loop = {
		array = TAK_painter_list
		value = v
		add_to_array = {
			array = TAK.TAK_painter_list_temp
			value = v
		}
	}
	#Then we find the highest in this temp array, and do this for as many brushes as there are
	for_loop_effect = {
		end = TAK_brush_count
		find_highest_in_array = {
			array = TAK_painter_list_temp
		}
		if = {
			limit = { check_variable = { v > 0 } }
			set_variable = { TAK_painter_list_temp_best_painter_index = i }
			set_variable = { TAK_painter_list_temp_best_painter_power = v }
			#We add their painting power to the total
			add_to_variable = { TAK_painting_power = TAK_painter_list_temp_best_painter_power }
			#And we set this element to 0 so that the next pass looks for the second highest
			set_variable = { TAK_painter_list_temp^TAK_painter_list_temp_best_painter_index = 0 }
		}
	}
	#Remove misc usage into account
	set_variable = { TAK_usable_painting_power = TAK_painting_power }
	subtract_from_variable = { TAK_usable_painting_power = modifier@TAK_painting_power_usage }
	clamp_variable = { var = TAK_usable_painting_power min = 0 max = 1000000 }
	clamp_variable = { var = TAK_painting_power min = 0 max = 1000000 }
}

TAK_clear_painter_flags = {
	if = {
		limit = { has_country_flag = TAK_painters_make_soldiers }
		clr_country_flag = TAK_painters_make_soldiers
	}
	if = {
		limit = { has_country_flag = TAK_painters_make_propaganda }
		clr_country_flag = TAK_painters_make_propaganda
	}
	if = {
		limit = { has_country_flag = TAK_painters_make_war_propaganda }
		clr_country_flag = TAK_painters_make_war_propaganda
	}
	if = {
		limit = { has_country_flag = TAK_painters_make_charles }
		clr_country_flag = TAK_painters_make_charles
	}
}

TAK_calculate_painting_output = {
	set_variable = { TAK_stability_weekly = 0 }
	set_variable = { TAK_military_factory_donations = 0 }
	set_variable = { TAK_weekly_manpower = 0 }
	set_variable = { TAK_war_support_weekly = 0 }
	set_variable = { TAK_painting_charles_weekly = 0 }
	if = {
		limit = { has_country_flag = TAK_painters_make_soldiers }
		#Make the soldiers
		set_variable = { TAK_weekly_manpower = TAK_weekly_manpower_per_painting_power }
		multiply_variable = { TAK_weekly_manpower = TAK_usable_painting_power }

		#Make the factories
		set_variable = { TAK_military_factory_donations = TAK_military_factory_donations_per_painting_power }
		multiply_variable = { TAK_military_factory_donations = TAK_usable_painting_power }
		round_variable = TAK_military_factory_donations
	}

	if = {
		limit = { has_country_flag = TAK_painters_make_propaganda }
		set_variable = { TAK_stability_weekly = TAK_stability_weekly_per_painting_power }
		multiply_variable = { TAK_stability_weekly = TAK_usable_painting_power }
	}

	if = {
		limit = { has_country_flag = TAK_painters_make_war_propaganda }
		set_variable = { TAK_war_support_weekly = TAK_war_support_weekly_per_painting_power }
		multiply_variable = { TAK_war_support_weekly = TAK_usable_painting_power }
	}

	if = {
		limit = { has_country_flag = TAK_painters_make_charles }
		set_variable = { TAK_painting_charles_weekly = TAK_painting_charles_weekly_per_painting_power }
		multiply_variable = { TAK_painting_charles_weekly = TAK_usable_painting_power }
	}
}
