calculate_private_sector_modifiers = {
	set_variable = { consumer_goods_perc = modifier@consumer_goods_factor }
	add_to_variable = { consumer_goods_perc = 1 }
	multiply_variable = { consumer_goods_perc = modifier@consumer_goods_expected_value }
	
	clamp_variable = { var = consumer_goods_perc min = 0 max = 100 }

	set_variable = { ps_modifiers = 0 }

	add_to_variable = { ps_modifiers = consumer_goods_perc }
	multiply_variable = { ps_modifiers = 0.5 }

	multiply_variable = { ps_modifiers = modifier@lok_ps_productivity_factor }

	set_variable = { lok_ps_production_speed_buildings_factor = ps_modifiers }
	set_variable = { lok_ps_industrial_capacity_factory = ps_modifiers }
	set_variable = { lok_ps_production_factory_max_efficiency_factor = ps_modifiers }

	multiply_variable = { lok_ps_production_speed_buildings_factor = modifier@lok_ps_production_speed_buildings_factor }
	multiply_variable = { lok_ps_industrial_capacity_factory = modifier@lok_ps_industrial_capacity_factory_factor }
	multiply_variable = { lok_ps_production_factory_max_efficiency_factor = modifier@lok_ps_production_factory_max_efficiency_factor_factor }

	#the odds of each state gettting new infra
	set_variable = { lok_ps_infra_growth_odds_true = modifier@lok_ps_infra_growth_odds }
	multiply_variable = { lok_ps_infra_growth_odds_true = consumer_goods_perc }
	multiply_variable = { lok_ps_infra_growth_odds_true = modifier@lok_ps_productivity_factor }
	multiply_variable = { lok_ps_infra_growth_odds_true = 100 }

	#the percentage of growth to attain
	set_variable = { lok_ps_growth_target_true = modifier@lok_ps_growth_target }
	multiply_variable = { lok_ps_growth_target_true = modifier@lok_ps_productivity_factor }

	#the absolute amount of factories to make each year
	set_variable = { lok_ps_factory_growth_goal = num_of_military_factories }
	add_to_variable = { lok_ps_factory_growth_goal = num_of_civilian_factories }
	multiply_variable = { lok_ps_factory_growth_goal = consumer_goods_perc }
	multiply_variable = { lok_ps_factory_growth_goal = lok_ps_growth_target_true }
	round_variable = lok_ps_factory_growth_goal

	clamp_variable = { var = ps_modifiers min = 0 max = 999 }

	if = {
		limit = { check_variable = { modifier@lok_ps_no_effects = 1 } }
		set_variable = { lok_ps_production_speed_buildings_factor = 0 }
		set_variable = { lok_ps_industrial_capacity_factory = 0 }
		set_variable = { lok_ps_production_factory_max_efficiency_factor = 0 }
		set_variable = { lok_ps_infra_growth_odds_true = 0 }
		set_variable = { lok_ps_factory_growth_goal = 0 }
		set_variable = { lok_ps_growth_target_true = 0 }
	}
	else_if = {
		limit = { check_variable = { modifier@lok_ps_bad_effects = 1 } }
		multiply_variable = { lok_ps_production_speed_buildings_factor = -1 }
		multiply_variable = { lok_ps_industrial_capacity_factory = -1 }
		multiply_variable = { lok_ps_production_factory_max_efficiency_factor = -1 }
		set_variable = { lok_ps_infra_growth_odds_true = 0 }
		set_variable = { lok_ps_factory_growth_goal = 0 }
		set_variable = { lok_ps_growth_target_true = 0 }
	}
	else = {
		clamp_variable = { var = lok_ps_production_speed_buildings_factor min = 0 max = 999 }
		clamp_variable = { var = lok_ps_industrial_capacity_factory min = 0 max = 999 }
		clamp_variable = { var = lok_ps_production_factory_max_efficiency_factor min = 0 max = 999 }
		clamp_variable = { var = lok_ps_infra_growth_odds_true min = 0 max = 999 }
		clamp_variable = { var = lok_ps_factory_growth_goal min = 0 max = 999 }
		clamp_variable = { var = lok_ps_growth_target_true min = 0 max = 999 }
	}

	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = LOK_private_sector_modifiers } } }
		add_dynamic_modifier = { modifier = LOK_private_sector_modifiers }
	}
}

update_base_state_population = {
	custom_effect_tooltip = update_base_state_population_tt
	hidden_effect = {
		set_variable = { population_at_game_start = new_base_population_k }
		set_variable = { population_last_check = state_population_k }
		update_state_productivity_modifier = yes
		set_variable = { lok_min_manpower_before_collapse = state_population_k } #Determine the population before the state collapses into anarchy
		divide_variable = { lok_min_manpower_before_collapse = 20 } #If 50k people are left out of 1M then stuff collapses.
		multiply_variable = { state_prod_change_from_pop = -1 }
		add_to_variable = { lok_state_productivity_value = state_prod_change_from_pop }
		set_variable = { state_prod_change_from_pop = 0 }
	}
}

#remove all pop and add it to death count
kill_all_state_population = {
	custom_effect_tooltip = kill_all_state_population_tt
	hidden_effect = {
		add_to_variable = { lok_state_misc_deaths = state_population_k }
		add_to_variable = { lok_state_deaths = state_population_k }
		add_to_variable = { lok_state_misc_deaths_display = state_population_k }
		var:controller = {
			add_to_variable = { lok_country_misc_deaths = PREV.state_population_k }
			add_to_variable = { lok_country_deaths = PREV.state_population_k }
			add_to_variable = { global.lok_global_misc_deaths = PREV.state_population_k }
			add_to_variable = { global.lok_global_deaths = PREV.state_population_k }
		}
		var:controller = { save_current_manpower = yes }
		while_loop_effect = {
			limit = { check_variable = { state_population_k > 0 } }
			add_manpower = -1000000
		}
		var:controller = { restore_previous_manpower = yes }
	}
}

LOK_NEK_war_update_opinion = {
	remove_opinion_modifier = { target = KUS modifier = KUS_NEK_war_hate }
	remove_opinion_modifier = { target = KUS modifier = KUS_NEK_war_oppose }
	remove_opinion_modifier = { target = KUS modifier = KUS_NEK_war_leaning }
	remove_opinion_modifier = { target = KUS modifier = KUS_NEK_war_support }
	remove_opinion_modifier = { target = NEK modifier = KUS_NEK_war_hate }
	remove_opinion_modifier = { target = NEK modifier = KUS_NEK_war_oppose }
	remove_opinion_modifier = { target = NEK modifier = KUS_NEK_war_leaning }
	remove_opinion_modifier = { target = NEK modifier = KUS_NEK_war_support }
	if = {
		limit = {
			check_variable = { KUS_NEK_support_level = -2 }
		}
		add_opinion_modifier = { target = KUS modifier = KUS_NEK_war_hate }
		add_opinion_modifier = { target = NEK modifier = KUS_NEK_war_support }
	}
	else_if = {
		limit = {
			check_variable = { KUS_NEK_support_level = -1 }
		}
		add_opinion_modifier = { target = KUS modifier = KUS_NEK_war_oppose }
		add_opinion_modifier = { target = NEK modifier = KUS_NEK_war_leaning }
	}
	else_if = {
		limit = {
			check_variable = { KUS_NEK_support_level = 1 }
		}
		add_opinion_modifier = { target = KUS modifier = KUS_NEK_war_leaning }
		add_opinion_modifier = { target = NEK modifier = KUS_NEK_war_oppose }
	}
	else_if = {
		limit = {
			check_variable = { KUS_NEK_support_level = 2 }
		}
		add_opinion_modifier = { target = KUS modifier = KUS_NEK_war_support }
		add_opinion_modifier = { target = NEK modifier = KUS_NEK_war_hate }
	}
}

update_state_productivity_modifier = {
	if = {
		limit = { 
			is_space_state = no
			var:controller = { is_non_playable_country = no }
		}
		# POPULATION CALCS
		# We determine the difference in pop between now and the last check
		# Now we must see how much of a % change that is relative to game start pop
		# So we add 5 points of productivity!

		if = {
			limit = {
				NOT = { has_state_flag = KUS_total_automation }
			}
			# Say it was 120 and is now 125, game start pop was 100
			set_variable = { state_prod_change_from_pop_this_check = state_population_k }

			# 125 - 120 = 5
			subtract_from_variable = { state_prod_change_from_pop_this_check = population_last_check }

			# 5 / 100 = 0.05, so 5% change
			divide_variable = { state_prod_change_from_pop_this_check = population_at_game_start }

			add_to_variable = { state_prod_change_from_pop = state_prod_change_from_pop_this_check }

			set_variable = { population_last_check = state_population_k }

			# The value of prod
			add_to_variable = { lok_state_productivity_value = state_prod_change_from_pop_this_check }
		}

		#Convert the productivity value to its modifier
		set_variable = { lok_state_productivity_modifier = lok_state_productivity_value }
		subtract_from_variable = { lok_state_productivity_modifier = 1 }
		clamp_variable = { var = lok_state_productivity_modifier min = -1 max = 1 }

		# #0.2 x -1 = -0.2
		# #-0.4 x -1 = 0.4
		# multiply_variable = { lok_state_productivity_modifier = -0.5 }
		# clamp_variable = { var = lok_state_productivity_modifier min = -0.75 max = 0.25 } #ultimately shoving one million people into the factory wont make it more efficient
	}
	else = { set_variable = { lok_state_productivity_modifier = 0 } }
}

regenerate_state_productivity_modifier = {
	if = {
		limit = {
			NOT = { check_variable = { damaged_infra_level_this_week > 0 } }
			check_variable = { lok_state_productivity_value < lok_state_productivity_value_base }
		}
		add_to_variable = { lok_state_productivity_value = 0.0005 }
		add_to_variable = { state_prod_change_from_regen = 0.0005 }
		clamp_variable = { var = lok_state_productivity_modifier min = 0 max = 2 }
	}
	#Convert the productivity value to its modifier
	set_variable = { lok_state_productivity_modifier = lok_state_productivity_value }
	subtract_from_variable = { lok_state_productivity_modifier = 1 }
	clamp_variable = { var = lok_state_productivity_modifier min = -1 max = 1 }
}

	if = {
		limit = { has_dynamic_modifier = { modifier = lok_matching_species } }

	}

		limit = { has_dynamic_modifier = { modifier = lok_matching_minority } }
		remove_dynamic_modifier = { modifier = lok_matching_minority }
	}
	#Matching species = compliance buff
	meta_effect = {
		text = {
			if = {
				limit = { var:controller = { has_idea = var_species_[SPECIES_ID] } }
				add_dynamic_modifier = { modifier = lok_matching_species }

			else_if = {
				limit = { var:controller = { has_idea = var_species_[MINORITY_ID] } }
				add_dynamic_modifier = { modifier = lok_matching_minority }
			}
		}
		SPECIES_ID = "[?species]"
		MINORITY_ID = "[?minority]"
	}
}

calculate_strategic_resource_modifiers = {
	if = {
		limit = { has_dynamic_modifier = { modifier = lok_strategic_resource_modifier } }
		remove_dynamic_modifier = { modifier = lok_strategic_resource_modifier }
	}

	set_variable = { stratres_1_modifier = 2 } #He3
	set_variable = { stratres_2_modifier = -0.50 } #Gold!

	set_variable = { stratres_modifier_intensity = num_controlled_states }
	subtract_from_variable = { stratres_modifier_intensity = 1 }
	divide_variable = { stratres_modifier_intensity = 5 }
	add_to_variable = { stratres_modifier_intensity = 1 }

	#Count resources
	set_variable = { stratres_1_count = 0 } #He3
	set_variable = { stratres_2_count = 0 } #Gold!

	every_controlled_state = {
		if = {
			limit = { check_variable = { stratres = 1 } }
			add_to_variable = { ROOT.stratres_1_count = 1 }
		}
		if = {
			limit = { check_variable = { stratres = 2 } }
			add_to_variable = { ROOT.stratres_2_count = 1 }
		}
	}

	multiply_variable = { stratres_1_modifier = stratres_1_count } #He3
	multiply_variable = { stratres_2_modifier = stratres_2_count } #Gold!

	divide_variable = { stratres_1_modifier = stratres_modifier_intensity }
	divide_variable = { stratres_2_modifier = stratres_modifier_intensity }

	if = {
		limit = { OR = { check_variable = { stratres_1_count > 0 } check_variable = { stratres_2_count > 0 } } }
		add_dynamic_modifier = { modifier = lok_strategic_resource_modifier }
	}
}

smart_remove_buildings = {
	# No errors yes yes.
	#
	# This effect will remove buildings in a state without the extra hassle of checking if they actually can remove the building (as it generates an error if there is no level of the building they're trying to remove)
	#
	# Call in state scope
	# EXPECTED VARIABLES
	# ?building_type = the building, duh. These are tokenizable so it makes my job a lot easier here
	# ?level_diff = levels removed, if set to 0 the script will assume you are using ?level_set
	# ?level_set = level set, use instead of ?level_diff if you want to set to a specific level (Dont set level_diff if you're using this)
	#
	# !!! IF NEITHER VARIABLE ARE SET, SCRIPT WILL ASSUME YOU ARE REMOVING ALL OF THOSE BUILDINGS IN THE STATE !!!
	#
	# Example use in script [State scope] (will remove up to 2 levels of civilian factories if capable. If there's 1 factory left it will only remove 1)
	#
	# set_temp_variable = { building_type = token:industrial_complex }
	# set_temp_variable = { level_diff = 2 }
	# smart_remove_buildings = yes

	# Calculating how many buildings to remove in this state
	if = {
		limit = { check_variable = { level_diff > 0 } }
		custom_effect_tooltip = smart_remove_buildings_tt1
	}
	else = { custom_effect_tooltip = smart_remove_buildings_tt2 }
	hidden_effect = {
		meta_effect = {
			text = {
				if = { # Only run if there's actually levels present. Causes Errors otherwise
					limit = { check_variable = { building_level@[BUILDING_TYPE] > 0 } }
					if = { # level_diff mode
						limit = { check_variable = { level_diff > 0 } }
						if = {
							limit = { # If the building level is greater/equal to the removed levels, just fire the effect
								check_variable = {
									var = building_level@[BUILDING_TYPE]
									value = level_diff
									compare = greater_than_or_equals
								}
							}
							remove_building = { type = [BUILDING_TYPE] level = level_diff }
						}
						# If building level is less than removed levels, set level_diff to the building level
						else = { set_temp_variable = { level_diff = building_level@[BUILDING_TYPE] } }
						remove_building = { type = [BUILDING_TYPE] level = level_diff }
					}
					else_if = { # Level Set Mode
						limit = { # Only run if the building level is higher than the desired level
							check_variable = {
								var = building_level@[BUILDING_TYPE]
								value = level_set
								compare = greater_than
							}
						}
						set_temp_variable = { levels_to_remove = building_level@[BUILDING_TYPE] }
						subtract_from_variable = { levels_to_remove = level_set }
						remove_building = { type = [BUILDING_TYPE] level = levels_to_remove }
					}
				}
			}
			BUILDING_TYPE = "[?building_type.GetTokenKey]"
		}
		# Cleanup! Lets you reuse the effect for other buildings in the same effect block
		set_temp_variable = { building_type = 0 }
		set_temp_variable = { level_set = 0 }
		set_temp_Variable = { level_diff = 0 }
		set_temp_Variable = { levels_to_remove = 0 }
	}
}

calculate_ipf_ksk_resistance = {
	if = {
		limit = { has_global_flag = katlinin_dead }
		clamp_variable = { var = ksk_strength min = 0 max = 3 }
	}
	else = { clamp_variable = { var = ksk_strength min = 0 max = 6 } }

	if = {
		limit = { has_global_flag = meowrius_dead }
		clamp_variable = { var = ipf_strength min = 0 max = 3 }
	}
	else = { clamp_variable = { var = ipf_strength min = 0 max = 6 } }

	clamp_variable = { var = ipf_opinion min = -5 max = 5 }
	clamp_variable = { var = ksk_opinion min = -5 max = 5 }
	if = {
		limit = { has_dynamic_modifier = { modifier = LOK_KSK_IPF_activity } }
		set_variable = { ipf_stability_factor = 0.025 }
		set_variable = { ipf_industrial_capacity_factory = 0.025 }
		set_variable = { ipf_resistance_activity = -0.05 }
		set_variable = { ipf_resistance_growth = -0.05 }
		set_variable = { ipf_democratic_drift = 0.01 }

		multiply_variable = { ipf_stability_factor = ipf_strength }
		multiply_variable = { ipf_industrial_capacity_factory = ipf_strength }
		multiply_variable = { ipf_resistance_activity = ipf_strength }
		multiply_variable = { ipf_resistance_growth = ipf_strength }
		multiply_variable = { ipf_democratic_drift = ipf_strength }
		
		if = {
			limit = { check_variable = { ipf_opinion < 0 } }
			multiply_variable = { ipf_stability_factor = -1 }
			multiply_variable = { ipf_industrial_capacity_factory = -1 }
			multiply_variable = { ipf_resistance_activity = -1 }
			multiply_variable = { ipf_resistance_growth = -1 }
		}

		else_if = {
			limit = { check_variable = { ipf_opinion = 0 } }
			multiply_variable = { ipf_stability_factor = 0 }
			multiply_variable = { ipf_industrial_capacity_factory = 0 }
			multiply_variable = { ipf_resistance_activity = 0 }
			multiply_variable = { ipf_resistance_growth = 0 }
		}

		set_variable = { ksk_war_support_factor = 0.025 }
		set_variable = { ksk_production_factory_max_efficiency_factor = 0.025 }
		set_variable = { ksk_compliance_gain = 0.005 }
		set_variable = { ksk_communism_drift = 0.01 }

		multiply_variable = { ksk_war_support_factor = ksk_strength }
		multiply_variable = { ksk_production_factory_max_efficiency_factor = ksk_strength }
		multiply_variable = { ksk_compliance_gain = ksk_strength }
		multiply_variable = { ksk_communism_drift = ksk_strength }

		if = {
			limit = { check_variable = { ksk_opinion < 0 } }
			multiply_variable = { ksk_war_support_factor = -1 }
			multiply_variable = { ksk_production_factory_max_efficiency_factor = -1 }
			multiply_variable = { ksk_compliance_gain = -1 }
		}

		else_if = {
			limit = { check_variable = { ksk_opinion = 0 } }
			multiply_variable = { ksk_war_support_factor = 0 }
			multiply_variable = { ksk_production_factory_max_efficiency_factor = 0 }
			multiply_variable = { ksk_compliance_gain = 0 }
		}
	}
}
calculate_ai_nuke_production = {
	set_variable = { factories_for_nukes = num_of_military_factories }
	multiply_variable = { factories_for_nukes = 0.05 } #5% of mils are put on nukes, say 100 so now it's 20
	if = { #Fascists like WMDs
		limit = { has_fascist_gov_group = yes }
		multiply_variable = { factories_for_nukes = 1.5 }
	}
	if = { #Katzens like WMDs
		limit = { is_herzlands_warlord = yes }
		multiply_variable = { factories_for_nukes = 2 }
	}
	if = { #Nations that REALLY like WMDs
		limit = { OR = { tag = NEU tag = AKR } }
		multiply_variable = { factories_for_nukes = 2 }
	}
	round_variable = factories_for_nukes
	set_variable = { nukes_this_month = factories_for_nukes }
	divide_variable = { nukes_this_month = 3 } #3 factories will make a nuke per month. 20/3 = 6.67 so 7 nukes
	#Bonus from factory output
	set_temp_variable = { nukes_bonus_factory_output = modifier@industrial_capacity_factory }
	add_to_temp_variable = { nukes_bonus_factory_output = 1 }
	#Bonus from efficiency cap
	set_temp_variable = { nukes_bonus_efficiency_cap = modifier@production_efficiency_cap_factor }
	multiply_temp_variable = { nukes_bonus_efficiency_cap = 0.5 }
	add_to_temp_variable = { nukes_bonus_efficiency_cap = 1 }
	#Add factory output + efficiency cap
	multiply_variable = { nukes_this_month = nukes_bonus_factory_output }
	multiply_variable = { nukes_this_month = nukes_bonus_efficiency_cap }
	if = { #If we can make less than 1 then 50% of making none
		limit = { check_variable = { nukes_this_month < 1 } }
		random_list = {
			1 = { set_variable = { nukes_this_month = 1 } }
			1 = { set_variable = { nukes_this_month = 0 } }
		}
	}
	round_variable = nukes_this_month
	for_loop_effect = {
		end = nukes_this_month
		random_list = {
			1 = {
				add_equipment_to_stockpile = { type = lok_fission_nuke_equipment_0 amount = 1 }
			}
			0 = {
				add_equipment_to_stockpile = { type = lok_fusion_nuke_equipment_0 amount = 1 }
				modifier = { add = 2 has_tech = fusion_nuke }
			}
			0 = {
				add_equipment_to_stockpile = { type = lok_cobalt_nuke_equipment_0 amount = 1 }
				modifier = { add = 1 has_tech = dirty_nuke }
			}
			0 = {
				add_equipment_to_stockpile = { type = lok_sturmer_nuke_equipment amount = 1 }
				modifier = { add = 1 has_tech = ultra_heavy_armor_tech }
			}
		}
	}
	multiply_variable = { factories_for_nukes = -1 }
	add_dynamic_modifier = { modifier = LOK_nukes_factories }
}

calculate_salvo_capacity = {
	set_temp_variable = { lok_salvo_capacity_added_last_week = lok_total_salvo_capacity } #Say 100
	set_variable = { lok_total_salvo_capacity = 0 }
			
	every_controlled_state = { add_to_variable = { PREV.lok_total_salvo_capacity = non_damaged_building_level@rocket_site } }
	multiply_variable = { lok_total_salvo_capacity = 10 } #10 nukes per silo

	set_temp_variable = { salvo_capacity_from_subs = num_ships_with_type@ballistic_submarine }
	multiply_temp_variable = { salvo_capacity_from_subs = 20 } #10 nukes per ballistic sub
	add_to_variable = { lok_total_salvo_capacity = salvo_capacity_from_subs }

	set_temp_variable = { salvo_capacity_from_riesigers = num_battalions_with_type@ultra_heavy_armor }
	multiply_temp_variable = { salvo_capacity_from_riesigers = 50 } #50 nukes per Riesiger
	add_to_variable = { lok_total_salvo_capacity = salvo_capacity_from_riesigers }

	#Great Missile Array wonder bonus
	if = {
		limit = {
			controls_state = 275
			275 = { check_variable = { wonder = 23 } }
		}
		if = {
			limit = { 275 = { is_core_of = PREV } }
			add_to_variable = { lok_total_salvo_capacity = 50 }
		}
		else_if = {
			limit = { 275 = { is_capital = yes } }
			add_to_variable = { lok_total_salvo_capacity = 100 }
		}
		else = {
			add_to_variable = { lok_total_salvo_capacity = 25 }
		}
	}


	if = {
		limit = { is_ai = yes }
		set_variable = { lok_total_salvo_capacity = 100 }
	}

	#Now let's say it's 120
	subtract_from_temp_variable = { lok_salvo_capacity_added_last_week = lok_total_salvo_capacity } #-20
	multiply_temp_variable = { lok_salvo_capacity_added_last_week = -1 } #20, we add that to the current capacity
	clamp_temp_variable = { var = lok_salvo_capacity_added_last_week min = 0 }
	add_to_variable = { lok_current_salvo_capacity = lok_salvo_capacity_added_last_week } #Add this to the current capacity
	clamp_variable = { var = lok_current_salvo_capacity max = lok_total_salvo_capacity }
}

calculate_total_food_consumption = {
	set_variable = { country_food_stockpile_max = modifier@lok_food_stockpile_capacity } #reset food stockpile capacity
	multiply_variable = { country_food_stockpile_max = modifier@lok_food_stockpile_factor } #Apply modifiers to stockpile capacity
	set_variable = { country_food_consumption = 0 } #Food that is being eaten
	set_variable = { country_food_balance = 0 }

	# 1. Fetch current Net Balance from the engine (This still has LAST week's consumption subtracted)
	set_variable = { country_food_balance = resource_produced@food } 
	set_variable = { country_food_imports = resource_imported@food } 
	add_to_variable = { country_food_balance = country_food_imports } 

	# 2. Calculate TRUE Gross Production by adding back LAST week's consumption modifier
	# We do this before overwriting the modifier so the math perfectly zeroes out the engine delay
	set_variable = { country_food_production = country_food_balance }
	add_to_variable = { country_food_production = country_food_consumption_modifier }

	# 3. Calculate NEW consumption from current territory
	every_controlled_state = { #Fetch how much food each state eats.
		calculate_food_consumption_in_state = yes
		PREV = {
			add_to_variable = { country_food_consumption = PREV.state_food_consumption }
		}
	}

	# The country modifier
	multiply_variable = { country_food_consumption = modifier@lok_country_food_consumption_factor }

	# 4. Update the modifier for the engine to use on the next tick
	set_variable = { country_food_consumption_modifier = country_food_consumption }
	# (Your commented-out excavation tech logic can safely stay here if you re-enable it later)
	round_variable = country_food_consumption_modifier

	# 5. Calculate the CORRECT New Balance for the rest of this script to use
	set_variable = { country_food_balance = country_food_production }
	subtract_from_variable = { country_food_balance = country_food_consumption }

	#FOOD STOCKPILING

	if = { #If balance is positive put into the stockpile
		limit = { check_variable = { country_food_balance > 0 } }
		set_temp_variable = { country_food_going_into_stockpile = country_food_balance }
		multiply_temp_variable = { country_food_going_into_stockpile = 0.5 } #as the monke said not all food can be put into storage
		add_to_variable = { country_food_stockpile = country_food_going_into_stockpile }
		clamp_variable = {  var = country_food_stockpile min = 0 max = country_food_stockpile_max } #Cant go over limit
		round_variable = country_food_stockpile
	}
	else_if = { #If balance is negative take from the stockpile if it exists
		limit = { check_variable = { country_food_balance < 0 } check_variable = { country_food_stockpile > 0 } }
		add_to_variable = { country_food_stockpile = country_food_balance }
		clamp_variable = {  var = country_food_stockpile min = 0 max = 1000000 } #Cant go below 0
		round_variable = country_food_stockpile
	}

	#CALCULATE % THAT IS MET - This is what determines the actual effects of food deficit
	set_variable = { country_food_needs_met_perc = country_food_production }
	divide_variable = { country_food_needs_met_perc = country_food_consumption }
	if = { #Nekeox and such
		limit = { check_variable = { country_food_consumption = 0 } }
		set_variable = { country_food_needs_met_perc = 1 }
	}
	set_variable = { country_food_needs_modifier = country_food_needs_met_perc } #For the modifier
	multiply_variable = { country_food_needs_met_perc = 100 }
	round_variable = country_food_needs_met_perc

	#MODIFIER
	subtract_from_variable = { country_food_needs_modifier = 1 }

	multiply_variable = { country_food_needs_modifier = 0.75 }

	if = {
		limit = { is_ai = yes }
		multiply_variable = { country_food_needs_modifier = 0.75 } #AI is a bit chtewpid innit
	}

	#Round a bit
	multiply_variable = { country_food_needs_modifier = 100 }
	round_variable = country_food_needs_modifier
	divide_variable = { country_food_needs_modifier = 100 }

	LOK_alert_lower_1 = yes
	LOK_alert_lower_2 = yes

	#Apply the modifiers
	if = {
		limit = { check_variable = { country_food_needs_met_perc < 50 } check_variable = { country_food_stockpile < 1 } }
		add_dynamic_modifier = { modifier = lok_starvation_deficit_modifier }
		if = {
			limit = { has_dynamic_modifier = { modifier = lok_food_deficit_modifier } }
			remove_dynamic_modifier = { modifier = lok_food_deficit_modifier }
		}
	}
	else_if = {
		limit = { check_variable = { country_food_needs_met_perc < 100 } check_variable = { country_food_stockpile < 1 } }
		add_dynamic_modifier = { modifier = lok_food_deficit_modifier }
		if = {
			limit = { has_dynamic_modifier = { modifier = lok_starvation_deficit_modifier } }
			remove_dynamic_modifier = { modifier = lok_starvation_deficit_modifier }
		}
		if = { #slight alert
			limit = { check_variable = { country_food_needs_met_perc < 75 } }
			LOK_alert_raise_1 = yes
		}
	}
	else = {
		if = {
			limit = { has_dynamic_modifier = { modifier = lok_food_deficit_modifier } }
			remove_dynamic_modifier = { modifier = lok_food_deficit_modifier }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = lok_starvation_deficit_modifier } }
			remove_dynamic_modifier = { modifier = lok_starvation_deficit_modifier }
		}
	}

	# if = {
	# 	limit = { check_variable = { country_food_needs_met_perc < 61 } check_variable = { country_food_stockpile < 1 } }
	# 	add_ideas = lok_famine_idea
	# }
	# else_if = {
	# 	limit = { check_variable = { country_food_needs_met_perc < 76 } check_variable = { country_food_stockpile < 1 } }
	# 	add_ideas = lok_severe_food_shortage_idea
	# }
	# else_if = {
	# 	limit = { check_variable = { country_food_needs_met_perc < 86 } check_variable = { country_food_stockpile < 1 } }
	# 	add_ideas = lok_mild_food_shortage_idea
	# }
	# else_if = {
	# 	limit = { check_variable = { country_food_needs_met_perc < 96 } check_variable = { country_food_stockpile < 1 } }
	# 	add_ideas = lok_small_food_shortage_idea
	# }
	# else_if = {
	# 	limit = { check_variable = { country_food_needs_met_perc > 119 } }
	# 	add_ideas = lok_food_surplus
	# }

	#IF % MET IS BELOW 61% AND STOCKPILES ARE EMPTY... PEOPLE DIE! 3% of the population will die every week until food goes back above 60%
	if = {
		limit = {
			check_variable = { country_food_needs_met_perc < 50 }
			check_variable = { country_food_stockpile < 1 }
			date > "100.1.5"
		}
		save_current_manpower = yes
		every_controlled_state = {
			set_variable = { lok_starvation_death_count = state_population_k }
			multiply_variable = { lok_starvation_death_count = 0.03 } #3% of the pop perishes
			round_variable = lok_starvation_death_count

			#Variable shenanigans to bypass overflow issues
			set_temp_variable = { lok_starvation_death_count_divided_10 = lok_starvation_death_count }
			multiply_temp_variable = { lok_starvation_death_count_divided_10 = -100 } #Divide by 10 and multiply by 1000 to get chunks that are equal to 1/10ths of the final casualty count - should be below overflow
			for_loop_effect = { #Now do it 10 times
				end = 10
				add_manpower = lok_starvation_death_count_divided_10
			}
			#And of course... the casualty counter
			add_to_variable = { lok_state_starvation_deaths = lok_starvation_death_count }
			add_to_variable = { lok_state_deaths = lok_starvation_death_count }

			var:CONTROLLER = {
				add_to_variable = { lok_country_starvation_deaths = PREV.lok_starvation_death_count }
				add_to_variable = { lok_country_deaths = PREV.lok_starvation_death_count }
				add_to_variable = { global.lok_global_starvation_deaths = PREV.lok_starvation_death_count }
				add_to_variable = { global.lok_global_deaths = PREV.lok_starvation_death_count }
			}
		}
		restore_previous_manpower = yes
	}
}

#weekly check to reduce and spread radiation
calculate_radiation_in_state = {
	if = { #Katown Pyramid protects!
		limit = { check_variable = { wonder = 1 } }
		set_variable = { lok_radiation_level = 0 }
	}
	if = {
		limit = { check_variable = { lok_radiation_level > 0 } }
		if = { #Below 0.2 it just goes away
			limit = { check_variable = { lok_radiation_level < 0.2 } }
			set_variable = { lok_radiation_level = 0 }
		}
		#Collapse into anarchy if radiation gets high enough
		if = {
			limit = { check_variable = { lok_radiation_level > 50 } }
			collapse_into_anarchy = yes
		}
		multiply_variable = { lok_radiation_level = 0.75 } #Half life of one week
		clamp_variable = { var = lok_radiation_level min = 0 max = 100000 }
		#Random spread to neighboring states: every week there is a 50% chance that a neighboring state get contaminated with half of the fallout.
		set_variable = { lok_radiation_spread = lok_radiation_level }
		multiply_variable = { lok_radiation_spread = 0.50 }
		random_neighbor_state = {
			random_list = {
				50 = {
					add_to_variable = { var = lok_radiation_level value = PREV.lok_radiation_spread }
					PREV = { subtract_from_variable = { lok_radiation_level = lok_radiation_spread } }
				}
				50 = {}
			}
		}
	}
}

#applied immediately when radiation is added - debuffs the state
calculate_radiation_state_modifiers = {
	if = { #Katown Pyramid protects!
		limit = { NOT = { check_variable = { wonder = 1 } } }
		#Prod loss from rads
		#1 rad = 0.1 prod loss, so divide by 1000
		set_variable = { state_prod_change_from_radiation_this_check = lok_added_radiation_level }
		divide_variable = { state_prod_change_from_radiation_this_check = -500 }
		add_to_variable = { state_prod_change_from_radiation = state_prod_change_from_radiation_this_check }
		add_to_variable = { lok_state_productivity_value = state_prod_change_from_radiation_this_check }
	}
	if = { #Katown Pyramid protects!
		limit = { check_variable = { wonder = 1 } }
		set_variable = { lok_radiation_level = 0 }
	}
	#Apply radiation state modifiers. Calculated as a percentage, with 10 radiation = -100% to all.
	set_variable = { lok_radiation_modifier = lok_radiation_level } #Say 5
	divide_variable = { lok_radiation_modifier = 10 } #Now 0.5
	set_variable = { lok_radiation_army_attrition_modifier = lok_radiation_modifier } #For attrition
	multiply_variable = { lok_radiation_modifier = -1 } #Make negative
	set_variable = { lok_radiation_army_modifier = lok_radiation_modifier } #For armies, not affected by species
	#Species Effects THIS IS THE "EFFECTS FROM RADIATION" TRAIT
	multiply_variable = { lok_radiation_modifier = modifier@lok_species_radiation_effect_factor }
	multiply_variable = { lok_radiation_death_percent = var:CONTROLLER.modifier@lok_country_radiation_effect_factor }
	clamp_variable = { var = lok_radiation_army_attrition_modifier min = 0 max = 0.9 } #Clamp between 0 and 0.9
	clamp_variable = { var = lok_radiation_army_modifier min = -0.9 max = 0 } #Clamp between 0 and -0.9
	clamp_variable = { var = lok_radiation_modifier min = -0.9 max = 0 } #Clamp between 0 and 0.9
}

#applied weekly, kill people due to radiation
apply_radiation_effects = {
	set_variable = { lok_radiation_threshold = modifier@lok_species_radiation_threshold_value }
	add_to_variable = { lok_radiation_threshold = var:CONTROLLER.modifier@lok_country_radiation_threshold_value }
	if = { #Kill people in the state
		limit = { #THIS IS THE "BASE RADIATION RESISTANCE TRAIT"
			check_variable = { lok_radiation_level > lok_radiation_threshold }
		}
		#Kill people in the state (RIP Bozo)
		set_variable = { lok_radiation_death_percent = lok_radiation_level }
		#Every level of radiation kills %, so 10 radiation = 10% death (Or 0.02)
		divide_variable = { lok_radiation_death_percent = 500 }

		#THIS IS THE "CASUALTIES FROM RADIATION" TRAIT
		multiply_variable = { lok_radiation_death_percent = modifier@lok_species_radiation_casualties_factor }
		#multiply_variable = { lok_radiation_death_percent = CONTROLLER.modifier@lok_country_radiation_casualties_factor } #FIX LATER

		set_variable = { lok_radiation_death_count = state_population_k }
		multiply_variable = { lok_radiation_death_count = lok_radiation_death_percent }
		clamp_variable = { var = lok_radiation_death_count min = 0 max = 2000 } #Avoid overflow
		multiply_variable = { lok_radiation_death_count = -1000 } #Turn to raw manpower
		var:controller = { save_current_manpower = yes }
		add_manpower = lok_radiation_death_count #Remove the people
		var:controller = { restore_previous_manpower = yes }
		divide_variable = { lok_radiation_death_count = -1000 } #Turn back to positive and divide by 1000 for tracking

		#Add to the trackers
		add_to_variable = { lok_state_radiation_deaths = lok_radiation_death_count }
		add_to_variable = { lok_state_deaths = lok_radiation_death_count }

		var:CONTROLLER = {
			add_to_variable = { lok_country_radiation_deaths = PREV.lok_radiation_death_count }
			add_to_variable = { lok_country_deaths = PREV.lok_radiation_death_count }
			add_to_variable = { global.lok_global_radiation_deaths = PREV.lok_radiation_death_count }
			add_to_variable = { global.lok_global_deaths = PREV.lok_radiation_death_count }
		}
	}
}

apply_wonder_effects = {
	#First we gotta reset all the effects!

	#Resetting the country level modifier array
	clear_array = wonder_modifiers_values
	resize_array = {
		array = wonder_modifiers_values
		size = @wonder_modifier_count
	}

	#Resetting the Wonder Dynamic Modifier In event the Controller doesnt control a modifier now!
	if = {
		limit = { has_dynamic_modifier = { modifier = lok_wonder_modifier } }
		remove_dynamic_modifier = { modifier = lok_wonder_modifier }
	}

	every_controlled_state = {
		limit = { check_variable = { wonder > 0 } }
		controller = {
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = lok_wonder_modifier } } }
				add_dynamic_modifier = { modifier = lok_wonder_modifier }
			}
		}
		for_each_loop = {
			array = wonder_modifiers_values
			index = i
			value = v
			break = break
			set_variable = { wonder_modifier_id = i }
			set_variable = { wonder_modifier_value = v }
			set_variable = { wonder_modifier_output_mult = 1 }
			add_to_variable = { wonder_modifier_output_mult = var:controller:modifier@lok_wonder_output_mult }
			meta_effect = {
				text = {
					if = {
						limit = {
							check_variable = {
								var = wonder_modifier_value
								value = 0
								compare = not_equals
							}
						}
						set_variable = {
							wonder_modifier_[WONDER_MODIFIER_ID] = [WONDER_MODIFIER_VALUE]
						}
						if = {
							limit = { is_capital = yes }
							multiply_variable = { wonder_modifier_[WONDER_MODIFIER_ID] = 2 }
						}
						else_if = {
							limit = {
								NOT = {
									is_core_of = var:controller
								}
							}
							multiply_variable = { wonder_modifier_[WONDER_MODIFIER_ID] = 0.5 }
						}
						multiply_variable = { wonder_modifier_[WONDER_MODIFIER_ID] = [WONDER_OUTPUT_MULT] }
						set_variable = {
							wonder_modifiers_values_real^[WONDER_MODIFIER_ID] = wonder_modifier_[WONDER_MODIFIER_ID] #Setting the post calculation modifier for the state tooltip
						}
						var:controller = {
							add_to_variable = {
								wonder_modifiers_values^[WONDER_MODIFIER_ID] = PREV.wonder_modifier_[WONDER_MODIFIER_ID]
							}
						}
					}
				}
				WONDER_MODIFIER_ID = "[?wonder_modifier_id]"
				WONDER_MODIFIER_VALUE = "[?wonder_modifier_value]"
				WONDER_OUTPUT_MULT = "[?wonder_modifier_output_mult]"
				debug = no
			}
		}
	}

restore_previous_manpower_function = {
	hidden_effect = {
		if = {
			limit = { has_variable = previous_country_manpower_se }
			set_variable = { current_country_manpower_se = manpower_k }
			#log = "previous_country_manpower_se: [?previous_country_manpower_se] current_country_manpower_se: [?current_country_manpower_se]"
			subtract_from_variable = { current_country_manpower_se = previous_country_manpower_se }
			#log = "difference: [?current_country_manpower_se]"
			while_loop_effect = {
				limit = { check_variable = { current_country_manpower_se > 99 } }
				subtract_from_variable = { current_country_manpower_se = 100 }
				add_manpower = -100000
				#log = "add -100000 manpower"
			}
			multiply_variable = { current_country_manpower_se = 1000 }
			round_variable = current_country_manpower_se
			multiply_variable = { current_country_manpower_se = -1 }
			#log = "add [?current_country_manpower_se] manpower"
			add_manpower = current_country_manpower_se
			#log = "current manpower [?manpower_k]"
			clear_variable = current_country_manpower_se
			#clear_variable = previous_country_manpower_se
		}
	}
}
