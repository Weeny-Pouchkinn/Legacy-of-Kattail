# State political pie chart support for Legacy of Kattail.
# Technique adapted from Yard1/HoI4-Scripted-GUI-Pie-Chart. Credit: Yard1.
# EXPECTED SCOPE: STATE
#
# AUTHORITATIVE INPUT:
#   lok_state_parties_array : 10 population shares, indices 0..8 ideologies, 9 apolitical.
# Values are expected to sum to 1.0; the renderer normalizes defensively.

LOK_rebuild_state_political_pie = {
    clear_array = lok_state_parties_normalized_array
    clear_array = lok_state_pie_chart_array
    set_temp_variable = { lok_state_pie_total = 0 }

    for_each_loop = {
        array = lok_state_parties_array
        value = lok_state_pie_raw_share
        index = lok_state_pie_i
        set_temp_variable = { lok_state_pie_share = lok_state_pie_raw_share }
        clamp_temp_variable = { var = lok_state_pie_share min = 0 max = 1 }
        add_to_temp_variable = { lok_state_pie_total = lok_state_pie_share }
    }

    if = {
        limit = { check_variable = { lok_state_pie_total > 0 } }
        set_temp_variable = { lok_state_pie_cumulative = 0 }

        for_each_loop = {
            array = lok_state_parties_array
            value = lok_state_pie_raw_share
            index = lok_state_pie_i

            set_temp_variable = { lok_state_pie_share = lok_state_pie_raw_share }
            clamp_temp_variable = { var = lok_state_pie_share min = 0 max = 1 }
            divide_temp_variable = { lok_state_pie_share = lok_state_pie_total }

            add_to_array = { array = lok_state_parties_normalized_array value = lok_state_pie_share }
            add_to_temp_variable = { lok_state_pie_cumulative = lok_state_pie_share }

            set_temp_variable = { lok_state_pie_target = lok_state_pie_cumulative }
            multiply_temp_variable = { lok_state_pie_target = 50 }
            round_temp_variable = lok_state_pie_target
            clamp_temp_variable = { var = lok_state_pie_target min = 0 max = 50 }

            set_temp_variable = { lok_state_pie_frame = lok_state_pie_i }
            add_to_temp_variable = { lok_state_pie_frame = 1 }

            while_loop_effect = {
                limit = { check_variable = { lok_state_pie_chart_array^num < lok_state_pie_target } }
                add_to_array = { array = lok_state_pie_chart_array value = lok_state_pie_frame }
            }
        }

        resize_array = { array = lok_state_pie_chart_array value = 10 size = 50 }
    }
    else = {
        resize_array = { array = lok_state_parties_normalized_array value = 0 size = 9 }
        add_to_array = { array = lok_state_parties_normalized_array value = 1 }
        resize_array = { array = lok_state_pie_chart_array value = 10 size = 50 }
    }

    # Cache the two highest ideologies for the dominant-ideology map mode.
    set_variable = { lok_state_dominant_ideology = 0 }
    set_variable = { lok_state_second_ideology = 1 }
    set_variable = { lok_state_dominant_share = -1 }
    set_variable = { lok_state_second_share = -1 }

    for_each_loop = {
        array = lok_state_parties_normalized_array
        value = lok_state_pie_share
        index = lok_state_pie_i

        if = {
            limit = { check_variable = { lok_state_pie_share > lok_state_dominant_share } }
            set_variable = { lok_state_second_ideology = lok_state_dominant_ideology }
            set_variable = { lok_state_second_share = lok_state_dominant_share }
            set_variable = { lok_state_dominant_ideology = lok_state_pie_i }
            set_variable = { lok_state_dominant_share = lok_state_pie_share }
        }
        else_if = {
            limit = { check_variable = { lok_state_pie_share > lok_state_second_share } }
            set_variable = { lok_state_second_ideology = lok_state_pie_i }
            set_variable = { lok_state_second_share = lok_state_pie_share }
        }
    }

    add_to_variable = { lok_state_political_pie_dirty = 1 }
}
