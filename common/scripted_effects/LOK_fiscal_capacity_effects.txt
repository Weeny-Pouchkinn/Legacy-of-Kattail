# Fiscal Capacity calculation and initialization.
# Modifier totals remain authoritative; variables mirror them for display and dynamic effects.

LOK_update_fiscal_capacity = {
	set_variable = { lok_fiscal_capacity_value = modifier@lok_fiscal_capacity }
	set_variable = { lok_fiscal_capacity_used = modifier@lok_fiscal_capacity_usage }

	set_variable = { lok_fiscal_capacity_remaining = lok_fiscal_capacity_value }
	subtract_from_variable = { lok_fiscal_capacity_remaining = lok_fiscal_capacity_used }

	set_variable = { lok_fiscal_capacity_overage = lok_fiscal_capacity_used }
	subtract_from_variable = { lok_fiscal_capacity_overage = lok_fiscal_capacity_value }
	clamp_variable = {
		var = lok_fiscal_capacity_overage
		min = 0
	}

	set_variable = { lok_fiscal_capacity_pp_penalty = lok_fiscal_capacity_overage }
	multiply_variable = { lok_fiscal_capacity_pp_penalty = -0.01 }

	if = {
		limit = {
			NOT = {
				has_dynamic_modifier = {
					modifier = lok_fiscal_capacity_overextension_modifier
				}
			}
		}
		add_dynamic_modifier = {
			modifier = lok_fiscal_capacity_overextension_modifier
		}
	}
	force_update_dynamic_modifier = yes
}
