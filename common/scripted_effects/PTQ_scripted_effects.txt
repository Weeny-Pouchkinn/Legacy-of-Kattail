# Pawtterston Quarries - Megacorp Scripted Effects
# Corps: 1=Pfotenstahl, 2=Bergkralle, 3=Quarzkatz, 4=Schnurrbart, 5=Samtpfote, 6=Nordlicht

PTQ_initialize_corps = {
	# Set initial opinions (0-100, starting at 50)
	set_variable = { ptq_pfotenstahl_opinion = 50 }
	set_variable = { ptq_bergkralle_opinion = 50 }
	set_variable = { ptq_quarzkatz_opinion = 50 }
	set_variable = { ptq_schnurrbart_opinion = 50 }
	set_variable = { ptq_samtpfote_opinion = 50 }
	set_variable = { ptq_nordlicht_opinion = 50 }

	# Initialize modifier variables to 0
	set_variable = { PTQ_pfotenstahl_industry = 0 }
	set_variable = { PTQ_pfotenstahl_arms_speed = 0 }
	set_variable = { PTQ_bergkralle_stability = 0 }
	set_variable = { PTQ_bergkralle_aluminium = 0 }
	set_variable = { PTQ_quarzkatz_research = 0 }
	set_variable = { PTQ_quarzkatz_efficiency = 0 }
	set_variable = { PTQ_schnurrbart_consumer = 0 }
	set_variable = { PTQ_schnurrbart_line_efficiency = 0 }
	set_variable = { PTQ_samtpfote_pp = 0 }
	set_variable = { PTQ_samtpfote_supply = 0 }
	set_variable = { PTQ_nordlicht_construction = 0 }
	set_variable = { PTQ_nordlicht_war_support = 0 }

	# Add the single unified dynamic modifier
	add_dynamic_modifier = { modifier = PTQ_megacorp_modifier }

	# Set initialized flag
	set_country_flag = PTQ_corps_initialized
}

# Count how many states each corp controls
PTQ_count_corp_states = {
	# Pfotenstahl AG - Winston (481), Ost-Winston (70)
	set_variable = { ptq_pfotenstahl_states = 0 }
	if = {
		limit = { 481 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_pfotenstahl_states = 1 }
	}
	if = {
		limit = { 70 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_pfotenstahl_states = 1 }
	}

	# Bergkralle Extraktion - Kattenburg (37)
	set_variable = { ptq_bergkralle_states = 0 }
	if = {
		limit = { 37 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_bergkralle_states = 1 }
	}

	# Quarzkatz Mineralien - Deaktiviertekamera (54), Magnetischergeschwollen (565)
	set_variable = { ptq_quarzkatz_states = 0 }
	if = {
		limit = { 54 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_quarzkatz_states = 1 }
	}
	if = {
		limit = { 565 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_quarzkatz_states = 1 }
	}

	# Schnurrbart Fertigung - Gligglepaw (88), Boogenstadt (566)
	set_variable = { ptq_schnurrbart_states = 0 }
	if = {
		limit = { 88 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_schnurrbart_states = 1 }
	}
	if = {
		limit = { 566 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_schnurrbart_states = 1 }
	}

	# Samtpfote Handelsgesellschaft - Ezikat (564), Middenpaw (567)
	set_variable = { ptq_samtpfote_states = 0 }
	if = {
		limit = { 564 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_samtpfote_states = 1 }
	}
	if = {
		limit = { 567 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_samtpfote_states = 1 }
	}

	# Nordlicht Energiewerke - Finstratus (568)
	set_variable = { ptq_nordlicht_states = 0 }
	if = {
		limit = { 568 = { is_controlled_by = ROOT } }
		add_to_variable = { ptq_nordlicht_states = 1 }
	}
}

# Recalculate all corp modifier variables based on opinion and state count
# Step 1: temp_factor = ((opinion - 50) / 50) * state_count
# Step 2: each modifier variable = temp_factor * base_value
PTQ_recalculate_all_corps = {
	PTQ_count_corp_states = yes

	# Pfotenstahl AG: industry + arms speed
	set_variable = { temp_factor = ptq_pfotenstahl_opinion }
	subtract_from_variable = { temp_factor = 50 }
	divide_variable = { temp_factor = 50 }
	multiply_variable = { temp_factor = ptq_pfotenstahl_states }

	set_variable = { PTQ_pfotenstahl_industry = temp_factor }
	multiply_variable = { PTQ_pfotenstahl_industry = 0.05 }
	set_variable = { PTQ_pfotenstahl_arms_speed = temp_factor }
	multiply_variable = { PTQ_pfotenstahl_arms_speed = 0.03 }

	# Bergkralle Extraktion: stability + aluminium
	set_variable = { temp_factor = ptq_bergkralle_opinion }
	subtract_from_variable = { temp_factor = 50 }
	divide_variable = { temp_factor = 50 }
	multiply_variable = { temp_factor = ptq_bergkralle_states }

	set_variable = { PTQ_bergkralle_stability = temp_factor }
	multiply_variable = { PTQ_bergkralle_stability = 0.04 }
	set_variable = { PTQ_bergkralle_aluminium = temp_factor }
	multiply_variable = { PTQ_bergkralle_aluminium = 15 }

	# Quarzkatz Mineralien: research + max efficiency
	set_variable = { temp_factor = ptq_quarzkatz_opinion }
	subtract_from_variable = { temp_factor = 50 }
	divide_variable = { temp_factor = 50 }
	multiply_variable = { temp_factor = ptq_quarzkatz_states }

	set_variable = { PTQ_quarzkatz_research = temp_factor }
	multiply_variable = { PTQ_quarzkatz_research = 0.03 }
	set_variable = { PTQ_quarzkatz_efficiency = temp_factor }
	multiply_variable = { PTQ_quarzkatz_efficiency = 0.03 }

	# Schnurrbart Fertigung: consumer goods (negative = good) + line efficiency
	set_variable = { temp_factor = ptq_schnurrbart_opinion }
	subtract_from_variable = { temp_factor = 50 }
	divide_variable = { temp_factor = 50 }
	multiply_variable = { temp_factor = ptq_schnurrbart_states }

	set_variable = { PTQ_schnurrbart_consumer = temp_factor }
	multiply_variable = { PTQ_schnurrbart_consumer = -0.02 }
	set_variable = { PTQ_schnurrbart_line_efficiency = temp_factor }
	multiply_variable = { PTQ_schnurrbart_line_efficiency = 0.03 }

	# Samtpfote Handelsgesellschaft: PP + supply (negative = good)
	set_variable = { temp_factor = ptq_samtpfote_opinion }
	subtract_from_variable = { temp_factor = 50 }
	divide_variable = { temp_factor = 50 }
	multiply_variable = { temp_factor = ptq_samtpfote_states }

	set_variable = { PTQ_samtpfote_pp = temp_factor }
	multiply_variable = { PTQ_samtpfote_pp = 0.05 }
	set_variable = { PTQ_samtpfote_supply = temp_factor }
	multiply_variable = { PTQ_samtpfote_supply = -0.02 }

	# Nordlicht Energiewerke: construction + war support
	set_variable = { temp_factor = ptq_nordlicht_opinion }
	subtract_from_variable = { temp_factor = 50 }
	divide_variable = { temp_factor = 50 }
	multiply_variable = { temp_factor = ptq_nordlicht_states }

	set_variable = { PTQ_nordlicht_construction = temp_factor }
	multiply_variable = { PTQ_nordlicht_construction = 0.05 }
	set_variable = { PTQ_nordlicht_war_support = temp_factor }
	multiply_variable = { PTQ_nordlicht_war_support = 0.04 }

	# Zero out seceded corps
	if = {
		limit = { has_country_flag = PTQ_pfotenstahl_seceded }
		set_variable = { PTQ_pfotenstahl_industry = 0 }
		set_variable = { PTQ_pfotenstahl_arms_speed = 0 }
	}
	if = {
		limit = { has_country_flag = PTQ_bergkralle_seceded }
		set_variable = { PTQ_bergkralle_stability = 0 }
		set_variable = { PTQ_bergkralle_aluminium = 0 }
	}
	if = {
		limit = { has_country_flag = PTQ_quarzkatz_seceded }
		set_variable = { PTQ_quarzkatz_research = 0 }
		set_variable = { PTQ_quarzkatz_efficiency = 0 }
	}
	if = {
		limit = { has_country_flag = PTQ_schnurrbart_seceded }
		set_variable = { PTQ_schnurrbart_consumer = 0 }
		set_variable = { PTQ_schnurrbart_line_efficiency = 0 }
	}
	if = {
		limit = { has_country_flag = PTQ_samtpfote_seceded }
		set_variable = { PTQ_samtpfote_pp = 0 }
		set_variable = { PTQ_samtpfote_supply = 0 }
	}
	if = {
		limit = { has_country_flag = PTQ_nordlicht_seceded }
		set_variable = { PTQ_nordlicht_construction = 0 }
		set_variable = { PTQ_nordlicht_war_support = 0 }
	}
}

# Monthly opinion decay - all corps lose 2 opinion per month unless recently given concessions
PTQ_monthly_opinion_decay = {
	if = {
		limit = { NOT = { has_country_flag = PTQ_pfotenstahl_concession_recent } NOT = { has_country_flag = PTQ_pfotenstahl_seceded } }
		subtract_from_variable = { ptq_pfotenstahl_opinion = 2 }
		clamp_variable = { var = ptq_pfotenstahl_opinion min = 0 max = 100 }
	}
	if = {
		limit = { NOT = { has_country_flag = PTQ_bergkralle_concession_recent } NOT = { has_country_flag = PTQ_bergkralle_seceded } }
		subtract_from_variable = { ptq_bergkralle_opinion = 2 }
		clamp_variable = { var = ptq_bergkralle_opinion min = 0 max = 100 }
	}
	if = {
		limit = { NOT = { has_country_flag = PTQ_quarzkatz_concession_recent } NOT = { has_country_flag = PTQ_quarzkatz_seceded } }
		subtract_from_variable = { ptq_quarzkatz_opinion = 2 }
		clamp_variable = { var = ptq_quarzkatz_opinion min = 0 max = 100 }
	}
	if = {
		limit = { NOT = { has_country_flag = PTQ_schnurrbart_concession_recent } NOT = { has_country_flag = PTQ_schnurrbart_seceded } }
		subtract_from_variable = { ptq_schnurrbart_opinion = 2 }
		clamp_variable = { var = ptq_schnurrbart_opinion min = 0 max = 100 }
	}
	if = {
		limit = { NOT = { has_country_flag = PTQ_samtpfote_concession_recent } NOT = { has_country_flag = PTQ_samtpfote_seceded } }
		subtract_from_variable = { ptq_samtpfote_opinion = 2 }
		clamp_variable = { var = ptq_samtpfote_opinion min = 0 max = 100 }
	}
	if = {
		limit = { NOT = { has_country_flag = PTQ_nordlicht_concession_recent } NOT = { has_country_flag = PTQ_nordlicht_seceded } }
		subtract_from_variable = { ptq_nordlicht_opinion = 2 }
		clamp_variable = { var = ptq_nordlicht_opinion min = 0 max = 100 }
	}

	PTQ_recalculate_all_corps = yes
}

# Check if any corp has reached revolt threshold (opinion < 10)
PTQ_check_revolt = {
	if = {
		limit = {
			check_variable = { ptq_pfotenstahl_opinion < 10 }
			NOT = { has_country_flag = PTQ_pfotenstahl_seceded }
			NOT = { has_country_flag = PTQ_pfotenstahl_revolt_warned }
		}
		set_country_flag = PTQ_pfotenstahl_revolt_warned
		set_variable = { PTQ_revolting_corp = 1 }
		country_event = { id = PTQ_megacorp.20 days = 1 }
	}
	if = {
		limit = {
			check_variable = { ptq_bergkralle_opinion < 10 }
			NOT = { has_country_flag = PTQ_bergkralle_seceded }
			NOT = { has_country_flag = PTQ_bergkralle_revolt_warned }
		}
		set_country_flag = PTQ_bergkralle_revolt_warned
		set_variable = { PTQ_revolting_corp = 2 }
		country_event = { id = PTQ_megacorp.20 days = 1 }
	}
	if = {
		limit = {
			check_variable = { ptq_quarzkatz_opinion < 10 }
			NOT = { has_country_flag = PTQ_quarzkatz_seceded }
			NOT = { has_country_flag = PTQ_quarzkatz_revolt_warned }
		}
		set_country_flag = PTQ_quarzkatz_revolt_warned
		set_variable = { PTQ_revolting_corp = 3 }
		country_event = { id = PTQ_megacorp.20 days = 1 }
	}
	if = {
		limit = {
			check_variable = { ptq_schnurrbart_opinion < 10 }
			NOT = { has_country_flag = PTQ_schnurrbart_seceded }
			NOT = { has_country_flag = PTQ_schnurrbart_revolt_warned }
		}
		set_country_flag = PTQ_schnurrbart_revolt_warned
		set_variable = { PTQ_revolting_corp = 4 }
		country_event = { id = PTQ_megacorp.20 days = 1 }
	}
	if = {
		limit = {
			check_variable = { ptq_samtpfote_opinion < 10 }
			NOT = { has_country_flag = PTQ_samtpfote_seceded }
			NOT = { has_country_flag = PTQ_samtpfote_revolt_warned }
		}
		set_country_flag = PTQ_samtpfote_revolt_warned
		set_variable = { PTQ_revolting_corp = 5 }
		country_event = { id = PTQ_megacorp.20 days = 1 }
	}
	if = {
		limit = {
			check_variable = { ptq_nordlicht_opinion < 10 }
			NOT = { has_country_flag = PTQ_nordlicht_seceded }
			NOT = { has_country_flag = PTQ_nordlicht_revolt_warned }
		}
		set_country_flag = PTQ_nordlicht_revolt_warned
		set_variable = { PTQ_revolting_corp = 6 }
		country_event = { id = PTQ_megacorp.20 days = 1 }
	}
}

# Execute revolt - corp secedes and their modifiers zeroed out
PTQ_execute_revolt = {

		limit = { check_variable = { PTQ_revolting_corp = 1 } }
		set_country_flag = PTQ_pfotenstahl_seceded
		add_stability = -0.15
		add_war_support = -0.10
		add_popularity = { ideology = communism popularity = 0.05 }

	if = {
		limit = { check_variable = { PTQ_revolting_corp = 2 } }
		set_country_flag = PTQ_bergkralle_seceded

# After putting down a revolt, redistribute the defeated corp's lands to a random neighboring corp
PTQ_redistribute_revolt_lands = {
	# Pick a random non-seceded corp to receive the lands
	random_list = {
		1 = {
			modifier = { factor = 0 has_country_flag = PTQ_pfotenstahl_seceded }
			modifier = { factor = 0 check_variable = { PTQ_revolting_corp = 1 } }
			set_variable = { PTQ_receiving_corp = 1 }
		}
		1 = {
			modifier = { factor = 0 has_country_flag = PTQ_bergkralle_seceded }
			modifier = { factor = 0 check_variable = { PTQ_revolting_corp = 2 } }
			set_variable = { PTQ_receiving_corp = 2 }
		}
		1 = {
			modifier = { factor = 0 has_country_flag = PTQ_quarzkatz_seceded }
			modifier = { factor = 0 check_variable = { PTQ_revolting_corp = 3 } }
			set_variable = { PTQ_receiving_corp = 3 }
		}
		1 = {
			modifier = { factor = 0 has_country_flag = PTQ_schnurrbart_seceded }
			modifier = { factor = 0 check_variable = { PTQ_revolting_corp = 4 } }
			set_variable = { PTQ_receiving_corp = 4 }
		}
		1 = {
			modifier = { factor = 0 has_country_flag = PTQ_samtpfote_seceded }
			modifier = { factor = 0 check_variable = { PTQ_revolting_corp = 5 } }
			set_variable = { PTQ_receiving_corp = 5 }

		add_to_variable = { ptq_bergkralle_opinion = 15 }

		limit = { check_variable = { PTQ_receiving_corp = 3 } }
		add_to_variable = { ptq_quarzkatz_opinion = 15 }
		clamp_variable = { var = ptq_quarzkatz_opinion min = 0 max = 100 }
	}

		limit = { check_variable = { PTQ_receiving_corp = 4 } }
		add_to_variable = { ptq_schnurrbart_opinion = 15 }
		clamp_variable = { var = ptq_schnurrbart_opinion min = 0 max = 100 }
