add_ideology_to_coalition = {
	if = {
		limit = { check_variable = { pol_party_array^ideology_temp = 3 } }
		custom_effect_tooltip = add_ideology_to_coalition_from_hostile_effect
	}
	else = {
		custom_effect_tooltip = add_ideology_to_coalition_effect
	}
	hidden_effect = {
		add_to_variable = { lok_pol_party_indicator = 1 }
		set_variable = { pol_party_array^ideology_temp = 2 }
		LOK_calculate_coalition_stability = yes
	}
}

LOK_dissolve_coalition = {
	hidden_effect = {
		if = { limit = { check_variable = { pol_party_array^0 = 2 } } set_variable = { pol_party_array^0 = 1 } }
		if = { limit = { check_variable = { pol_party_array^1 = 2 } } set_variable = { pol_party_array^1 = 1 } }
		if = { limit = { check_variable = { pol_party_array^2 = 2 } } set_variable = { pol_party_array^2 = 1 } }
		if = { limit = { check_variable = { pol_party_array^3 = 2 } } set_variable = { pol_party_array^3 = 1 } }
		if = { limit = { check_variable = { pol_party_array^4 = 2 } } set_variable = { pol_party_array^4 = 1 } }
		if = { limit = { check_variable = { pol_party_array^5 = 2 } } set_variable = { pol_party_array^5 = 1 } }
		if = { limit = { check_variable = { pol_party_array^6 = 2 } } set_variable = { pol_party_array^6 = 1 } }
		if = { limit = { check_variable = { pol_party_array^7 = 2 } } set_variable = { pol_party_array^7 = 1 } }
		if = { limit = { check_variable = { pol_party_array^8 = 2 } } set_variable = { pol_party_array^8 = 1 } }
		add_to_variable = { lok_pol_party_indicator = 1 }
	}
}
