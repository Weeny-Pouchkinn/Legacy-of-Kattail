add_ideology_to_coalition = {
	if = {
		limit = { check_variable = { pol_party_array^ideology_temp = 3 } }
		custom_effect_tooltip = add_ideology_to_coalition_from_hostile_effect
	}
	else = {
		custom_effect_tooltip = add_ideology_to_coalition_effect
	}
	hidden_effect = {
		add_to_variable = { lok_pol_party_indicator = 1 }
		set_variable = { pol_party_array^ideology_temp = 2 }
	}
}

make_ideology_neutral = {
	if = {
		limit = { check_variable = { pol_party_array^ideology_temp = 2 } }
		custom_effect_tooltip = make_ideology_neutral_from_coalition_effect
	}
	else_if = {
		limit = { check_variable = { pol_party_array^ideology_temp = 3 } }
		custom_effect_tooltip = make_ideology_neutral_from_hostile_effect
	}
	else = {
		custom_effect_tooltip = make_ideology_neutral_effect
	}
	hidden_effect = {
		add_to_variable = { lok_pol_party_indicator = 1 }
		set_variable = { pol_party_array^ideology_temp = 1 }
	}
}

make_ideology_hostile = {
	if = {
		limit = { check_variable = { pol_party_array^ideology_temp = 2 } }
		custom_effect_tooltip = make_ideology_hostile_from_coalition_effect
	}
	else = {
		custom_effect_tooltip = make_ideology_hostile_effect
	}
	hidden_effect = {
		add_to_variable = { lok_pol_party_indicator = 1 }
		set_variable = { pol_party_array^ideology_temp = 3 }
	}
}
