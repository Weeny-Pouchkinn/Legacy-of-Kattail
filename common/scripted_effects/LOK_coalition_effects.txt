# Chnage party popularity across your coalition based on their respective popularity.

# Meant to simulate how the larger party likely has their candidate running the ministry responsible for this boost in popularity

	custom_effect_tooltip = LOK_change_coalition_party_popularity_weighted_tt
	# Calculating Coalition and Non Coalition Share
	if = { limit = { check_variable = { pol_party_array^0 = 1 } } add_to_temp_variable = { coalition_total_pop = party_popularity@communism } }
	if = { limit = { check_variable = { pol_party_array^1 = 1 } } add_to_temp_variable = { coalition_total_pop = party_popularity@socialism } }
	if = { limit = { check_variable = { pol_party_array^2 = 1 } } add_to_temp_variable = { coalition_total_pop = party_popularity@social_democratic } }
	if = { limit = { check_variable = { pol_party_array^3 = 1 } } add_to_temp_variable = { coalition_total_pop = party_popularity@social_liberal } }
	if = { limit = { check_variable = { pol_party_array^4 = 1 } } add_to_temp_variable = { coalition_total_pop = party_popularity@democratic } }
	if = { limit = { check_variable = { pol_party_array^5 = 1 } } add_to_temp_variable = { coalition_total_pop = party_popularity@social_conservative } }
