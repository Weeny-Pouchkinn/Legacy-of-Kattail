roq_count_votation_results = {
	set_variable = { roq_num_federal_approval_2 = 0 }
	set_variable = { roq_num_federal_approval_1 = 0 }
	set_variable = { roq_num_federal_approval_0 = 0 }
	set_variable = { roq_num_federal_approval_minus_1 = 0 }
	set_variable = { roq_num_federal_approval_minus_2 = 0 }
	for_each_scope_loop = {
		array = roq_fed_states_array
		if = {
			limit = { check_variable = { roq_federal_approval = 2 } }
			ROQ = { add_to_variable = { roq_num_federal_approval_2 = 1 } }
		}
		else_if = {
			limit = { check_variable = { roq_federal_approval = 1 } }
			ROQ = { add_to_variable = { roq_num_federal_approval_1 = 1 } }
		}
		else_if = {
			limit = { check_variable = { roq_federal_approval = 0 } }
			ROQ = { add_to_variable = { roq_num_federal_approval_0 = 1 } }
		}
		else_if = {
			limit = { check_variable = { roq_federal_approval = -1 } }
			ROQ = { add_to_variable = { roq_num_federal_approval_minus_1 = 1 } }
		}
		else_if = {
			limit = { check_variable = { roq_federal_approval = -2 } }
			ROQ = { add_to_variable = { roq_num_federal_approval_minus_2 = 1 } }
		}
	}
	set_variable = { roq_pro_federal = roq_num_federal_approval_1 }
	set_variable = { roq_anti_federal = roq_num_federal_approval_minus_1 }
	add_to_variable = { roq_pro_federal = roq_num_federal_approval_2 }
	add_to_variable = { roq_anti_federal = roq_num_federal_approval_minus_2 }
}
