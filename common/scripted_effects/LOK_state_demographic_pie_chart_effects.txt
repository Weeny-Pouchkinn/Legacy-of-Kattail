# Legacy of Kattail - selected-state species/culture pie chart support.
# EXPECTED SCOPE: STATE.
#
# AUTHORITATIVE INPUTS:
#   lok_state_species_array : 61 share entries (0..1)
#   lok_state_culture_array : 103 share entries (0..1)
#
# Render directly from numeric array elements via for_each_loop.

LOK_rebuild_state_species_pie = {
    clear_array = lok_state_species_pie_chart_array
    set_temp_variable = { lok_demo_total = 0 }

    for_each_loop = {
        array = lok_state_species_array
        value = lok_demo_raw_share
        index = lok_demo_i
        set_temp_variable = { lok_demo_share = lok_demo_raw_share }
        clamp_temp_variable = { var = lok_demo_share min = 0 max = 1 }
        add_to_temp_variable = { lok_demo_total = lok_demo_share }
    }

    if = {
        limit = { check_variable = { lok_demo_total > 0 } }
        set_temp_variable = { lok_demo_cumulative = 0 }
        set_temp_variable = { lok_demo_last_frame = 1 }

        for_each_loop = {
            array = lok_state_species_array
            value = lok_demo_raw_share
            index = lok_demo_i

            set_temp_variable = { lok_demo_share = lok_demo_raw_share }
            clamp_temp_variable = { var = lok_demo_share min = 0 max = 1 }

            if = {
                limit = { check_variable = { lok_demo_share > 0.000001 } }
                divide_temp_variable = { lok_demo_share = lok_demo_total }
                add_to_temp_variable = { lok_demo_cumulative = lok_demo_share }

                set_temp_variable = { lok_demo_target = lok_demo_cumulative }
                multiply_temp_variable = { lok_demo_target = 50 }
                round_temp_variable = lok_demo_target
                clamp_temp_variable = { var = lok_demo_target min = 0 max = 50 }

                set_temp_variable = { lok_demo_frame = lok_demo_i }
                add_to_temp_variable = { lok_demo_frame = 1 }
                set_temp_variable = { lok_demo_last_frame = lok_demo_frame }

                while_loop_effect = {
                    limit = { check_variable = { lok_state_species_pie_chart_array^num < lok_demo_target } }
                    add_to_array = { array = lok_state_species_pie_chart_array value = lok_demo_frame }
                }
            }
        }

        while_loop_effect = {
            limit = { check_variable = { lok_state_species_pie_chart_array^num < 50 } }
            add_to_array = { array = lok_state_species_pie_chart_array value = lok_demo_last_frame }
        }
    }
}
