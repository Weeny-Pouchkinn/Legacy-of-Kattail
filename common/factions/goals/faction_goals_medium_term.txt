	description = a_large_faction_goal_desc
	category = medium_term
	group = FOCUS_FILTER_POLITICAL_CHARACTER

		num_faction_members > 4
	}
	completed = {
		num_faction_members > 9
	}

		add_faction_power_projection = 200

		add_faction_initiative = 1
	}

		base = 1
	}
}

faction_goal_industrial_expansion = {
	name = industrial_expansion_goal_name
	description = industrial_expansion_goal_desc
	category = medium_term
	group = FOCUS_FILTER_INDUSTRY
	visible = {
		count_in_collection = {
			collection = { input = game:scope operators = { faction_members } name = FACTION_NAME }
			buildings = { industrial_complex }
			size > 49
		}
	}
	completed = {
		count_in_collection = {
			collection = { input = game:scope operators = { faction_members } name = FACTION_NAME }
			buildings = { industrial_complex }
			size > 149
		}
	}
	complete_effect = {
		add_faction_power_projection = 200
		custom_effect_tooltip = generic_skip_one_line_tt
		every_faction_member = {
			tooltip = AT_LEAST_50_CIVS
			visible_when_empty = yes
			limit = { num_of_civilian_factories > 49 }
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_industrial_expansion } }
				add_tech_bonus = { name = industrial_expansion_tech_bonus bonus = 0.5 uses = 2 category = industry }
			}
		}
	}
	ai_will_do = { base = 1 }
}
