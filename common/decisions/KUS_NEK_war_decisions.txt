				every_country = {

							array = global.KUS_NEK_war_supporters

					remove_opinion_modifier = { target = NEK modifier = KUS_NEK_war_leaning }
					remove_opinion_modifier = { target = NEK modifier = KUS_NEK_war_support }
				}
				clr_global_flag = KUS_NEK_war_active
				if = {
					limit = { has_dynamic_modifier = { modifier = KUS_international_support } }
					remove_dynamic_modifier = { modifier = KUS_international_support }
				}
				NEK = {
					if = {
						limit = { has_dynamic_modifier = { modifier = NEK_international_support } }
						remove_dynamic_modifier = { modifier = NEK_international_support }
					}
				}
			}
		}
