						transfer_units_fraction = {
							target = var:KEP_fascist_country
							stockpile_ratio = 0.40
							army_ratio = 0.40
							navy_ratio = 0.40
							air_ratio = 0.40
						}
						set_country_flag = KEP_civil_war_original
					}

					var:KEP_fascist_country = {
						save_global_event_target_as = KEP_fascist_splinter
						set_country_flag = KEP_fascist_splinter
						set_politics = {
							ruling_party = fascism
							elections_allowed = no
						}
						set_popularities = {
							fascism = 100
						}
						set_capital = { state = 1023 }
						declare_war_on = {
							target = KEP
							type = annex_everything
						}
					}
				}

				KEP = {
					add_civil_war_target = event_target:KEP_fascist_splinter
				}

				# The peace is only an armistice: Plaveta now fights both Queplerian factions.
				PLA = {
					declare_war_on = {
						target = KEP
						type = annex_everything
					}
					declare_war_on = {
						target = event_target:KEP_fascist_splinter
						type = annex_everything
					}
					activate_mission = PLA_seize_queplerian_land_mission
				}

				news_event = { id = LOK_KEP.3 hours = 6 }
			}
		}
