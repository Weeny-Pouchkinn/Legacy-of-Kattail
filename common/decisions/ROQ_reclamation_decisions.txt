ROQ_reclamation_cat = {

	# ─────────────────────────────────────────────────────────────────────
	# TIMED MISSION: Cloudstrider Intervention (MCF – has actual timeout effects)
	# ─────────────────────────────────────────────────────────────────────
	MCF_reclamation_cloudstrider_intervention = {
		icon = GFX_decision_hol_war_on_pacifism
		allowed = { tag = MCF }

		activation = {
			has_global_flag = ROQ_reclamation_active
			is_in_array = { array = global.ROQ_reclamation_ssu_supporters value = THIS }
		}

		available = { always = no }

		cancel_trigger = {
			NOT = { has_global_flag = ROQ_reclamation_active }
		}

		is_good = yes
		fire_only_once = yes
		selectable_mission = no
		days_mission_timeout = 3

		timeout_effect = {
			# If ROQ is still fighting anyone in the targets array, add them to MCF's faction
			every_country = {
				limit = {
					is_in_array = { array = global.ROQ_reclamation_targets value = THIS }
					country_exists = THIS
					has_war_with = ROQ
					MCF = { is_faction_leader = yes }
				}
				MCF = { add_to_faction = PREV }
				country_event = { id = ROQ_reclamation.20 }
			}
			# Cleanup: clear flag and arrays (modifier stays — it represents permanent training)
			hidden_effect = { ROQ_reclamation_cleanup = yes }
		}
	}

	# ─────────────────────────────────────────────────────────────────────
	# TIMED MISSION: Cloudstrider Intervention (everyone else – display only)
	# ─────────────────────────────────────────────────────────────────────
	ROQ_reclamation_cloudstrider_intervention = {
		icon = GFX_decision_hol_war_on_pacifism
		allowed = {
			OR = {
				tag = ROQ
				tag = FSR
				tag = DOM
				is_flusiobalkan = yes
			}
		}

		activation = {
			has_global_flag = ROQ_reclamation_active
			NOT = { tag = MCF }
			OR = {
				tag = ROQ
				is_in_array = { array = global.ROQ_reclamation_targets value = THIS }
				is_in_array = { array = global.ROQ_reclamation_ssu_supporters value = THIS }
			}
		}

		available = { always = no }

		cancel_trigger = {
			NOT = { has_global_flag = ROQ_reclamation_active }
		}

		is_good = no
		fire_only_once = yes
		selectable_mission = no
		days_mission_timeout = 180

		timeout_effect = { }
	}

	# ─────────────────────────────────────────────────────────────────────
	# FSR / DOM LOBBY DECISIONS
	# target_array = global.ROQ_reclamation_ssu_supporters
	# 50% success if only one of FSR/DOM exists; 33% if both exist.
	# ─────────────────────────────────────────────────────────────────────

	FSR_DOM_reclamation_lobby = {
		icon = GFX_decision_generic_diplomacy

		visible = {
			has_global_flag = ROQ_reclamation_active
			is_in_array = { array = global.ROQ_reclamation_targets value = THIS }
		}

		available = {
			hidden_trigger = {
				check_variable = { FROM.ROQ_reclamation_support_level < 4 }
			}
		}

		target_array = global.ROQ_reclamation_ssu_supporters

		target_trigger = {
			FROM = {
				exists = yes
				# Only allow lobbying nations that aren't already Adamant
			}
		}
 
		cost = 50
		days_remove = 30

		remove_effect = {
			custom_effect_tooltip = ROQ_reclamation_lobby_tt
			if = {
				# Both FSR and DOM are in the targets array → 33% (1 in 3) chance
				limit = {
					country_exists = FSR
					country_exists = DOM
					FSR = { is_in_array = { array = global.ROQ_reclamation_targets value = FSR } }
					DOM = { is_in_array = { array = global.ROQ_reclamation_targets value = DOM } }
				}
				random_list = {
					1 = {
						custom_effect_tooltip = ROQ_reclamation_lobby_success_tt
						hidden_effect = {
							FROM = {
								add_to_variable = { ROQ_reclamation_support_level = 1 }
								clamp_variable = { var = ROQ_reclamation_support_level min = 1 max = 4 }
							}
						}
					}
					2 = {
						custom_effect_tooltip = ROQ_reclamation_lobby_fail_tt
					}
				}
			}
			else = {
				# Only one target nation → 50% (1 in 2) chance
				random_list = {
					1 = {
						custom_effect_tooltip = ROQ_reclamation_lobby_success_tt
						hidden_effect = {
							FROM = {
								add_to_variable = { ROQ_reclamation_support_level = 1 }
								clamp_variable = { var = ROQ_reclamation_support_level min = 1 max = 4 }
							}
						}
					}
					1 = {
						custom_effect_tooltip = ROQ_reclamation_lobby_fail_tt
					}
				}
			}
		}

		ai_will_do = {
			base = 50
		}
	}
}
