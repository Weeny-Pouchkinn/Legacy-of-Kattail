	#Infiltrate state
	TAK_infiltrate_state = {
		icon = GFX_decision_eng_trade_unions_support

		allowed = {
			original_tag = TAK
		}

		state_target = yes
		on_map_mode = map_only
		target_array = TAK_possible_infiltration_states

		visible = {
			tag = TAK
			has_completed_focus = TAK_perfect_imitation
			NOT = { has_war_with = FROM.controller }
		}

		available = {
			FROM = { check_variable = { TAK_infiltration_level < 3 } }
			custom_trigger_tooltip = {
				tooltip = TAK_infiltrate_state_available_tt
				hidden_trigger = {
					check_variable = { TAK_painting_power > 24 }
				}
			}
		}

		ai_will_do = {
			factor = 10
		}

		days_remove = 30

		cost = 10

		modifier = {
			TAK_painting_power_usage = 25
		}

		complete_effect = {
			add_to_variable = { TAK_painting_power_usage = 25 }
			TAK_update_painting_stats = yes
		}

		remove_effect = {
			subtract_from_variable = { TAK_painting_power_usage = 25 }
			TAK_update_painting_stats = yes
			if = {
				limit = { has_DLC = "La Resistance" }
				add_intel = { target = FROM.controller army_intel = 5 }
				add_intel = {
					target = FROM.controller
					civilian_intel = 10
				}
			}
			FROM = {
				custom_effect_tooltip = TAK_infiltrate_state_effect_tt
				add_to_variable = { TAK_infiltration_level = 1 }
				#the more developed a state is, the more likely it is that this is found out
				set_variable = { TAK_threshold_to_halve_reveal_odds = var:owner.num_core_states }
				multiply_variable = { TAK_threshold_to_halve_reveal_odds = 10 } #if takicardie has 10 times the painting power than the nation has core states the odds are halved
				if = {
					limit = { check_variable = { TAK.TAK_painting_power < TAK_threshold_to_halve_reveal_odds } }
					if = {
						limit = { OR = { has_state_category = large_city has_state_category = metropolis has_state_category = megalopolis } }
						random_list = {
							10 = { var:controller = { country_event = { id = TAK_event.1003 } } }
							90 = {}
						}
					}
					else = {
						random_list = {
							5 = { var:controller = { country_event = { id = TAK_event.1003 } } }
							95 = {}
						}
					}				
				}
				else = { #tak does have the right odds
					if = {
						limit = { OR = { has_state_category = large_city has_state_category = metropolis has_state_category = megalopolis } }
						random_list = {
							5 = { var:controller = { country_event = { id = TAK_event.1003 } } }
							95 = {}
						}
					}
					else = {
						random_list = {
							2 = { var:controller = { country_event = { id = TAK_event.1003 } } }
							98 = {}
						}
					}		
				}
			}
		}

		cancel_trigger = {
			has_war_with = FROM.controller
		}

		cancel_effect = {
			subtract_from_variable = { TAK_painting_power_usage = 25 }
		}
	}
