						}

						}

			every_owned_state = {
				limit = {
					has_state_flag = TAK_puppet_core
				}
				
				# 1. We are in a STATE scope. Save this state so we can reference it later.
				save_event_target_as = tak_state_to_transfer
				
				# 2. Scope to countries to find the rightful owner
				random_country = {
					limit = {
						is_puppet_of = TAK
						
						# 3. Scope TO the saved state. 
						# Inside this block, PREV refers to the country we are checking in 'random_country'
						event_target:tak_state_to_transfer = {
							is_core_of = PREV
						}
					}
					# 4. We found the country. Save it as our target.
					save_event_target_as = tak_country_receiver
				}
				
				# 5. We are back in the STATE scope. Transfer it to the saved country.
				transfer_state_to = event_target:tak_country_receiver
			}
		}


		ai_will_do = {
			factor = 200
		}
	}
