
is_enemy_ideology = {
	OR = {
		AND = {
			has_communist_gov_group = yes
			ROOT = {
				OR = {
					has_fascist_gov_group = yes
					has_democratic_gov_group = yes
				}
			}
		}
		AND = {
			has_democratic_gov_group = yes
			ROOT = {
				OR = {
					has_communist_gov_group = yes
					has_fascist_gov_group = yes
				}
			}
		}
		AND = {
			has_fascist_gov_group = yes
			ROOT = {
				OR = {
					has_democratic_gov_group = yes
					has_communist_gov_group = yes
				}
			}
		}
	}
}

	}
}
