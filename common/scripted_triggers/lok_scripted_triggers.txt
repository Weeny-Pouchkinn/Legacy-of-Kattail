can_lose_democracy_support = {
	set_temp_variable = { far_left_total_popularity = party_popularity@communism }
	add_to_temp_variable = { far_left_total_popularity = party_popularity@socialism }
	check_variable = { far_left_total_popularity < 0.18 }

	fascism < 0.18

	neutrality < 0.18

	set_temp_variable = { democracy_total_popularity = party_popularity@democratic }
	add_to_temp_variable = { democracy_total_popularity = party_popularity@social_democratic }
	add_to_temp_variable = { democracy_total_popularity = party_popularity@social_liberal }
	add_to_temp_variable = { democracy_total_popularity = party_popularity@social_conservative }
	add_to_temp_variable = { democracy_total_popularity = party_popularity@authoritarian_democratic }
	check_variable = { democracy_total_popularity > 0.65 }
}
