# Parliament status helpers. The existing politics system uses 1=neutral, 2=coalition, 3=hostile.
LOK_parliament_party_is_hostile = {
	check_variable = { pol_party_array^lok_parliament_party_index = 3 }
}

LOK_parliament_party_is_coalition = {
	check_variable = { pol_party_array^lok_parliament_party_index = 2 }
}

LOK_parliament_party_is_ruling = {
	OR = {
		AND = {
			check_variable = { lok_parliament_party_index = 0 }
			has_government = communism
		}
		AND = {
			check_variable = { lok_parliament_party_index = 1 }
			has_government = socialism
		}
		AND = {
			check_variable = { lok_parliament_party_index = 2 }
			has_government = social_democratic
		}
		AND = {
			check_variable = { lok_parliament_party_index = 3 }
			has_government = social_liberal
		}
		AND = {
			check_variable = { lok_parliament_party_index = 4 }
			has_government = democratic
		}
		AND = {
			check_variable = { lok_parliament_party_index = 5 }
			has_government = social_conservative
		}
		AND = {
			check_variable = { lok_parliament_party_index = 6 }
			has_government = authoritarian_democratic
		}
		AND = {
			check_variable = { lok_parliament_party_index = 7 }
			has_government = neutrality
		}
		AND = {
			check_variable = { lok_parliament_party_index = 8 }
			has_government = fascism
		}
	}
}
