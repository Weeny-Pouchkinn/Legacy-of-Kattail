LOK_has_tribal_economy_lock = {
	has_idea = lok_tribal_economy
}

LOK_has_paluush_economy_lock = {
	has_idea = PLR_paluush_masteries
}

LOK_has_special_economy_lock = {
	OR = {
		LOK_has_tribal_economy_lock = yes
		LOK_has_paluush_economy_lock = yes
	}
}

				has_government = fascism
			}
		}
