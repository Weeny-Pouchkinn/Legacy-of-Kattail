# These are equipped-force objectives, not hard caps on divisions in training.
# has_army_size accounts for strength, so empty tank divisions do not fulfill it.
# Keep demand stable while the queue consumes equipment; reserves gate field
# conversion, not recruitment weights, to avoid cancelling partly equipped units.
lok_ai_land_regional_starter_armor = {
	enable = {
		lok_ai_land_regional_armor_country = yes
		has_army_size = { size < @regional_starter_goal type = armor }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = role_ratio
		id = armor
		value = @regional_armor_weight
	}
}

lok_ai_land_regional_expanded_armor = {

# Production floors are deliberately small; normal equipment needs still set
# the main budget. Specific archetypes prevent an armored-car line from meeting
# a truck requirement or an unrelated armored vehicle from meeting the tank floor.
lok_ai_land_rifle_production_floor = {
	enable = {
		lok_ai_land_conventional_country = yes
		num_of_military_factories > 0
		OR = {
			AND = {
				lok_ai_land_regular_economy = yes
				has_equipment = { infantry_equipment < @rifle_upgrade_buffer }
			}
			AND = {
				NOT = { lok_ai_land_regular_economy = yes }
				has_equipment = { infantry_equipment < @rifle_crisis_reserve }
			}
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = infantry_equipment
		value = 1
	}
}
