defined_text = {
    name = LOK_PrintParty1Demands

    text = {
        trigger = {
            # Copy the real array into a common temporary array
            clear_temp_array = lok_print_array

            all_of = {
                array = lok_party_1_demands_array
                value = val

                add_to_temp_array = {
                    lok_print_array = val
                }
            }

            # Start BEFORE element 0
            set_temp_variable = {
                lok_print_index = -1
            }
        }

        localization_key = "[LOK_PrintArray]"
    }
}

defined_text = {
    name = LOK_PrintArray

    text = {
        trigger = {
            # Move to next element
            add_to_temp_variable = {
                lok_print_index = 1
            }

            # Are we still inside the array?
            check_variable = {
                lok_print_index < lok_print_array^num
            }

            # Store the current element
            set_temp_variable = {
                lok_print_value = lok_print_array^lok_print_index
            }
        }

        localization_key = "LOK_print_array_element"
    }

    # End of array
    text = {
        localization_key = ""
    }
}
