defined_text = {
	name = LOK_GetParliamentCommunismBlock
	text = {
		trigger = {
			set_temp_variable = { lok_parliament_party_index = 0 }
			LOK_parliament_party_is_hostile = yes
		}
		localization_key = LOK_parliament_communism_hostile
	}
	text = {
		trigger = {
			set_temp_variable = { lok_parliament_party_index = 0 }
			AND = {
			LOK_parliament_party_is_hostile = no
			LOK_parliament_party_is_ruling = yes
		}
		}
		localization_key = LOK_parliament_communism_ruling
	}
	text = {
		trigger = {
			set_temp_variable = { lok_parliament_party_index = 0 }
			AND = {
			LOK_parliament_party_is_hostile = no
			LOK_parliament_party_is_ruling = no
			LOK_parliament_party_is_coalition = yes
		}
		}
		localization_key = LOK_parliament_communism_coalition
	}
	text = {
		trigger = {
			set_temp_variable = { lok_parliament_party_index = 0 }
			AND = {
			LOK_parliament_party_is_hostile = no
			LOK_parliament_party_is_ruling = no
			LOK_parliament_party_is_coalition = no

		}
		localization_key = LOK_parliament_communism_neutral
	}
}
