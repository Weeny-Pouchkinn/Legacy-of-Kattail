# ROQ Reclamation scripted localisation

# Returns a coloured label for THIS country's current support level
defined_text = {
	name = GetROQReclamationSupportLevel
	text = {
		trigger = { check_variable = { ROQ_reclamation_support_level > 3 } }
		localization_key = ROQ_reclamation_support_level_adamant
	}
	text = {
		trigger = { check_variable = { ROQ_reclamation_support_level > 2 } }
		localization_key = ROQ_reclamation_support_level_protective
	}
	text = {
		trigger = { check_variable = { ROQ_reclamation_support_level > 1 } }
		localization_key = ROQ_reclamation_support_level_supportive
	}
	text = {
		localization_key = ROQ_reclamation_support_level_disinterested
	}
}

# Returns a one-line description of what THIS country can do as an SSU supporter.
# Used in FSR/DOM's lobby decision description so players know what each ally offers.
defined_text = {
	name = GetROQReclamationWhatTheyCanDo
	text = {
		trigger = { tag = MCF }
		localization_key = ROQ_reclamation_what_mcf_can_do
	}
	text = {
		trigger = { tag = KHA }
		localization_key = ROQ_reclamation_what_kha_can_do
	}
	text = {
		trigger = { tag = SLR }
		localization_key = ROQ_reclamation_what_slr_can_do
	}
	text = {
		trigger = { tag = TEN }
		localization_key = ROQ_reclamation_what_ten_can_do
	}
	text = {
		localization_key = ROQ_reclamation_what_generic_can_do
	}
}

# Returns the current description of the Reclamation depending on the stage
defined_text = {
	name = GetROQReclamationStageDesc
	text = {
		trigger = {
			ROQ = { check_variable = { reclamation_stage = 2 } }
		}
		localization_key = ROQ_reclamation_stage_2_desc
	}
	text = {
		trigger = {
			ROQ = { check_variable = { reclamation_stage = 1 } }
		}
		localization_key = ROQ_reclamation_stage_1_desc
	}
	text = {
		localization_key = ROQ_reclamation_stage_inactive_desc
	}
}
