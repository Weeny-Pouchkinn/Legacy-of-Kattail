defined_text = {
    name = GetCulture1
    text = {
        trigger = {
            is_in_array = { array = global.culture_1_array value = THIS }
        }
        localization_key = culture_1_name
    }
    # Fallback if not in the array
    text = {
        localization_key = empty_string
    }
}
