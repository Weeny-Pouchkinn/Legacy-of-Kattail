defined_text = {
	name = GetNobleHouseStateOwnershipSatisfaction
	text = {
		trigger = {
			OR = {
				check_variable = { noble_house_states_array^substate_id > noble_house_happy_states_array^substate_id }
				check_variable = { noble_house_states_array^substate_id = noble_house_happy_states_array^substate_id }
			}
		}
		localization_key = noble_house_happy_states
	}
	text = {
		trigger = {
			check_variable = { noble_house_states_array^substate_id < noble_house_expected_states_array^substate_id }
		}
		localization_key = noble_house_unhappy_states
	}
	text = {
		trigger = {
			OR = {
				check_variable = { noble_house_states_array^substate_id < noble_house_happy_states_array^substate_id }
				check_variable = { noble_house_states_array^substate_id > noble_house_expected_states_array^substate_id }
				check_variable = { noble_house_states_array^substate_id = noble_house_expected_states_array^substate_id }
			}
		}
		localization_key = noble_house_normal_states
	}
}

defined_text = {
	name = GetNobleHouseHappyOrNot
	text = {
		trigger = {
			OR = {
				check_variable = { noble_house_opinion_array^substate_id > 0 }
				check_variable = { noble_house_opinion_array^substate_id = 0 }
			}
		}
		localization_key = noble_house_happy
	}
	text = {
		trigger = {
			check_variable = { noble_house_opinion_array^substate_id < 0 }
		}
		localization_key = noble_house_unhappy
	}
}
